    55	        }

    57	      } else {
    58	        for (const node of this.node.rule.nodes) {
    59	          yield* new GraphPointer(node, this).resolve();
    60	        }
    61	      }
    62	    } else if (isGraphPointerRuleEnd(this)) {

    64	        yield this;

   259	        this.pos = parseSpace(src, this.pos + 1, isNested);
   260	      } else {

   262	      }

   267	    const src = this.src;

   272	      this.checkDuration();

   275	      this.parseSequence(ruleName, rule, depth);
   276	    }
   277	    rule.push(new InternalRuleDefEnd());
