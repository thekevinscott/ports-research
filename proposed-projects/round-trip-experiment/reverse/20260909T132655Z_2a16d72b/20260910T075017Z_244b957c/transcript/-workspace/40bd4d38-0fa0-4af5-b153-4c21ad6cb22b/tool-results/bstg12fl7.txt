    27	class Graph:
    28	    def __init__(self, grammar: str, stacked_rules, root_id: int):
    29	        self.grammar = grammar
    30	        self.roots: Dict[int, RootNode] = {}
    31	        self.previous_code_points: List[int] = []
    32	
    33	        rule_refs = []
    34	        unique_rules = GenericSet(get_serialized_rule_key)
    35	        for stack_id, stack in enumerate(stacked_rules):
    36	            nodes: RootNode = {}
    37	            stack = stack or []
    38	            for path_id, path in enumerate(stack):
    39	                node: Optional[GraphNode] = None
    40	                for step_id in range(len(path) - 1, -1, -1):
    41	                    next_node = node
    42	                    rule = stack[path_id][step_id]
    43	                    unique_rules.add(rule)
    44	                    if is_rule_ref(rule):
    45	                        rule_refs.append(rule)
    46	                    # rules coming in may be identical but have different references.
    47	                    # here, we ensure we always use the same reference for an identical
    48	                    # rule. this makes future comparisons easier.
    49	                    unique_rule = unique_rules.get(rule)
    50	                    if unique_rule is None:
    51	                        raise Exception("Could not get unique rule")
    52	                    node = GraphNode(
    53	                        unique_rule, GraphNodeMeta(stack_id, path_id, step_id), next_node
    54	                    )
    55	                if node is None:
    56	                    raise Exception("Could not get node")
    57	                nodes[path_id] = node
    58	
    59	            self.roots[stack_id] = nodes
    60	
    61	        root_node = self.roots.get(root_id)
    62	        if not root_node:
    63	            raise Exception(f"Root node not found for value: {root_id}")
    64	        self._root_node = root_node
    65	
    66	        for rule_ref in rule_refs:
    67	            referenced_nodes = []
    68	            for node in self._get_root_node(rule_ref.value).values():
    69	                if node not in referenced_nodes:
    70	                    referenced_nodes.append(node)
    71	            rule_ref.nodes = referenced_nodes

    73	    def _get_root_node(self, value: int) -> RootNode:
    74	        root_node = self.roots.get(value)
    75	        if not root_node:
    76	            raise Exception(f"Root node not found for value: {value}")
    77	        return root_node

    79	    def _get_initial_pointers(self) -> GenericSet:
    80	        pointers = _make_pointers()
    81	
    82	        root_node = self._root_node
    83	        if not root_node:
    84	            raise Exception("Root node is not defined")
    85	
    86	        for node, parent in self._fetch_nodes_for_root_node(root_node):
    87	            pointer = GraphPointer(node, parent)
    88	            for resolved_pointer in self._resolve_pointer(pointer):
    89	                pointers.add(resolved_pointer)
    90	        return pointers

    92	    @staticmethod
    93	    def _set_valid(pointers: List[GraphPointer], valid: bool) -> None:
    94	        for pointer in pointers:
    95	            pointer.valid = valid
    96	
    97	    def _parse(self, current_pointers: GenericSet, code_point: int) -> GenericSet:
    98	        for rule, graph_pointers in self._iterate_over_pointers(current_pointers):
    99	            if is_rule_char(rule):
   100	                valid = False
   101	                for possible_code_point in rule.value:
   102	                    if valid:
   103	                        break
   104	                    if is_range(possible_code_point):
   105	                        valid = is_point_in_range(code_point, possible_code_point)
   106	                    else:
   107	                        valid = code_point == possible_code_point
   108	                self._set_valid(graph_pointers, valid)
   109	            elif is_rule_char_excluded(rule):
   110	                valid = True
   111	                for possible_code_point in rule.value:
   112	                    if not valid:
   113	                        break
   114	                    if is_range(possible_code_point):
   115	                        valid = not is_point_in_range(code_point, possible_code_point)
   116	                    else:
   117	                        valid = code_point != possible_code_point
   118	                self._set_valid(graph_pointers, valid)
   119	            elif not is_rule_end(rule):
   120	                raise Exception(f"Unsupported rule: {rule!r}")
   121	
   122	        # a pointer's id is the sum of its node's id and its parent's id chain.
   123	        # if two pointers share the same id, it means they point to the same node and
   124	        # have identical parent chains. for the purposes of walking the graph, we only
   125	        # need to keep one of them.
   126	        next_pointers = _make_pointers()
   127	        for current_pointer in current_pointers:
   128	            for unresolved_next_pointer in current_pointer.fetch_next():
   129	                for resolved_next_pointer in self._resolve_pointer(
   130	                    unresolved_next_pointer
   131	                ):
   132	                    next_pointers.add(resolved_next_pointer)
   133	        return next_pointers
   134	
   135	    def _resolve_pointer(
   136	        self, unresolved_pointer: GraphPointer
   137	    ) -> Iterator[GraphPointer]:
   138	        for resolved_pointer in unresolved_pointer.resolve():
   139	            if is_rule_ref(resolved_pointer.node.rule):
   140	                raise Exception(
   141	                    "Encountered a reference rule when building pointers to the graph"
   142	                )
   143	            if is_rule_end(resolved_pointer.node.rule) and resolved_pointer.parent:
   144	                raise Exception(
   145	                    "Encountered an ending rule with a parent when building pointers "
   146	                    "to the graph"
   147	                )
   148	            yield resolved_pointer
   149	
   150	    def add(self, src: ValidInput, _pointers: Optional[GenericSet] = None) -> GenericSet:
   151	        pointers = _pointers if _pointers is not None else self._get_initial_pointers()
   152	        code_points = get_input_as_code_points(src)
   153	        for code_point_pos, code_point in enumerate(code_points):
   154	            pointers = self._parse(pointers, code_point)
   155	            if pointers.size == 0:
   156	                raise InputParseError(
   157	                    code_points, code_point_pos, list(self.previous_code_points)
   158	                )
   159	        self.previous_code_points.extend(code_points)
   160	        return pointers
   161	
   162	    def _fetch_nodes_for_root_node(
   163	        self,
   164	        root_nodes: RootNode,
   165	        parent: Optional[GraphPointer] = None,
   166	    ) -> Iterator[Tuple[GraphNode, Optional[GraphPointer]]]:
   167	        """Yield either the node, or if a reference rule, the referenced node.
   168	
   169	        This is distinct from the logic in GraphPointer because that needs a rule ref
   170	        with already defined nodes; this function is used to _set_ those nodes.
   171	        """
   172	        for node in root_nodes.values():
   173	            if is_rule_ref(node.rule):
   174	                yield from self._fetch_nodes_for_root_node(
   175	                    self._get_root_node(node.rule.value), GraphPointer(node, parent)
   176	                )
   177	            else:
   178	                yield node, parent
   179	
   180	    def print(self, pointers: Optional[GenericSet] = None, colors: bool = False) -> str:
   181	        col = colorize if colors else no_color
   182	        graph_view: List[str] = []
   183	        for nodes in self.roots.values():
   184	            for node in nodes.values():
   185	                graph_view.append(
   186	                    node.print(pointers=pointers, show_position=True, colorize=col)
   187	                )
   188	        joined = "\n".join(graph_view)
   189	        return f"\n{joined}"
   190	
   191	    def __repr__(self) -> str:
   192	        return self.print(colors=True)
   193	
   194	    @staticmethod
   195	    def _iterate_over_pointers(
   196	        pointers: GenericSet,
   197	    ) -> Iterator[Tuple[object, List[GraphPointer]]]:
   198	        # Rules are deduplicated by reference when the graph is built, so grouping by
   199	        # object identity keeps distinct-but-equal rules apart the way the reference
   200	        # implementation's Map does.
   201	        seen_rules: Dict[int, Tuple[object, List[GraphPointer]]] = {}
   202	        for pointer in pointers:
   203	            rule = pointer.rule
   204	            if is_rule_ref(rule):
   205	                raise Exception(
   206	                    "Encountered a reference rule in the graph, this should not happen"
   207	                )
   208	            entry = seen_rules.get(id(rule))
   209	            if entry is None:
   210	                seen_rules[id(rule)] = (rule, [pointer])
   211	            else:
   212	                entry[1].append(pointer)
   213	
   214	        for rule, graph_pointers in seen_rules.values():
   215	            yield rule, graph_pointers

    14	class GraphPointer:
    15	    def __init__(self, node, parent: Optional["GraphPointer"] = None):
    16	        if node is None:
    17	            raise Exception("Node is undefined")
    18	        self.node = node
    19	        self.parent = parent
    20	        self.id = f"{parent.id}-{node.id}" if parent else node.id
    21	        self._valid: Optional[bool] = None
    22	
    23	    def resolve(self, resolved: bool = False) -> Iterator["GraphPointer"]:
    24	        """
    25	        1. If the current node is an end node, and the pointer has a parent, return the
    26	           parent's `fetch_next`; else return nothing.
    27	        2. If the current node is a rule ref, yield the referenced nodes, _unless_
    28	           resolved is true, in which case it returns next.
    29	        3. If the current node is a char or range, we go to the next node. If none
    30	           exists, throw an error.
    31	        """
    32	        if is_graph_pointer_rule_ref(self):
    33	            if resolved:
    34	                if self.node.next is None:
    35	                    raise Exception(f"No next node: {self.node!r}")
    36	                yield from GraphPointer(self.node.next, self.parent).resolve()
    37	            else:
    38	                for node in self.node.rule.nodes:
    39	                    yield from GraphPointer(node, self).resolve()
    40	        elif is_graph_pointer_rule_end(self):
    41	            if not self.parent:
    42	                yield self
    43	            else:
    44	                yield from self.parent.resolve(True)
    45	        elif is_graph_pointer_rule_char(self) or is_graph_pointer_rule_char_exclude(
    46	            self
    47	        ):
    48	            yield self
    49	        else:
    50	            raise Exception(f"Unknown rule: {self.node.rule!r}")
    51	
    52	    def fetch_next(self) -> Iterator["GraphPointer"]:
    53	        # if this pointer is invalid, then we don't return any new pointers
    54	        if self._valid is False:
    55	            return
    56	
    57	        # if this pointer is an end node, we return the parent's next node. If no parent
    58	        # exists, we return nothing, since it's the end of the line.
    59	        if is_rule_end(self.node.rule):
    60	            if self.parent:
    61	                yield from self.parent.fetch_next()
    62	        else:
    63	            if self.node.next is None:
    64	                raise Exception(f"No next node: {self.node!r}")
    65	            yield from GraphPointer(self.node.next, self.parent).resolve()
    66

    68	    def rule(self):
    69	        return self.node.rule
    70	
    71	    @property

    77	        self._valid = valid

    23	def build_rule_stack(linear_rules: Optional[List]) -> List[List]:
    24	    paths: List = []
    25	    stack: List[List] = []
    26	
    27	    linear_rules = linear_rules or []
    28	
    29	    idx = 0
    30	    while idx < len(linear_rules):
    31	        rule_def = linear_rules[idx]
    32	        if is_rule_def_char(rule_def) or is_rule_def_char_not(rule_def):
    33	            # this could be a single char, or a range, or a sequence of alts; we don't
    34	            # know until we step through it.
    35	            char_rule = _make_char_rule(rule_def)
    36	            idx += 1
    37	            rule = linear_rules[idx] if idx < len(linear_rules) else None
    38	            while idx < len(linear_rules) and (
    39	                is_rule_def_char_rng_upper(rule) or is_rule_def_char_alt(rule)
    40	            ):
    41	                if is_rule_def_char_rng_upper(rule):
    42	                    # previous rule value should be a number
    43	                    prev_value = char_rule.value.pop() if char_rule.value else None
    44	                    if is_range(prev_value):
    45	                        raise Exception(
    46	                            "Unexpected range, expected a number but got an array: "
    47	                            f"{prev_value!r}"
    48	                        )
    49	                    if prev_value is None:
    50	                        raise Exception("Unexpected undefined value")
    51	                    char_rule.value.append([prev_value, rule.value])
    52	                if is_rule_def_char_alt(rule):
    53	                    char_rule.value.append(rule.value)
    54	                idx += 1
    55	                rule = linear_rules[idx] if idx < len(linear_rules) else None
    56	            paths.append(char_rule)
    57	        else:
    58	            if is_rule_def_alt(rule_def):
    59	                if not paths:
    60	                    raise Exception("Encountered alt without anything before it")
    61	                paths.append(RuleEnd())
    62	                stack.append(paths)
    63	                paths = []
    64	            elif is_rule_def_end(rule_def):
    65	                paths.append(RuleEnd())
    66	            elif is_rule_def_ref(rule_def):
    67	                paths.append(RuleRef(rule_def.value))
    68	            elif is_rule_def_char_alt(rule_def):
    69	                raise Exception(
    70	                    "Encountered char alt, should be handled by above block: "
    71	                    f"{rule_def.type}"
    72	                )
    73	            else:
    74	                raise Exception(f"Unsupported rule type: {rule_def.type}")
    75	            idx += 1
    76	
    77	    if not is_rule_end(paths[-1] if paths else None):
    78	        paths.append(RuleEnd())
    79	
    80	    stack.append(paths)
    81

    29	        raise GrammarParseError(
    30	            src, pos, "Unexpected end of grammar input, failed to complete parse"
    31	        )
    32	    return ord(src[pos]), 1

    30	    def _parse(self, src: str) -> None:
    31	        # move cursor forward until we reach non-whitespace content
    32	        self.pos = parse_space(src, 0, True)
    33	
    34	        while self.pos < len(src):
    35	            self._parse_rule(src)
    36	
    37	        # Validate the state to ensure that all rules are defined
    38	        for rule in self.rules:
    39	            if rule is None:
    40	                continue
    41	            for elem in rule:
    42	                if elem.type == InternalRuleType.RULE_REF:
    43	                    # Ensure that the rule at that location exists
    44	                    rule_exists = (
    45	                        elem.value < len(self.rules)
    46	                        and self.rules[elem.value] is not None
    47	                        and len(self.rules[elem.value]) > 0
    48	                    )
    49	                    if not rule_exists:
    50	                        missing_rule_name = self.symbol_ids.reverse_get(elem.value)
    51	                        missing_rule_pos = self.symbol_ids.get_pos(missing_rule_name)
    52

    54	                        while missing_rule_pos < len(src) and (
    55	                            src[missing_rule_pos] in (":", "=")
    56	                            or _WHITESPACE.search(src[missing_rule_pos])
    57	                        ):

    61	                            src,
    62	                            missing_rule_pos,
    63	                            f'Undefined rule identifier "{missing_rule_name}"',
    64	                        )

    69	        rule_id = self.get_symbol_id(name, len(name))
    70	
    71	        # Skip over whitespace characters and find the ::= sequence
    72	        self.pos = parse_space(src, self.pos, True)
    73	        if not (
    74	            char_at(src, self.pos) == ":"
    75	            and char_at(src, self.pos + 1) == ":"
    76	            and char_at(src, self.pos + 2) == "="
    77	        ):

   116	    def parse_sequence(
   117	        self,
   118	        rule_name: str,
   119	        out_elements: List[InternalRuleDef],
   120	        depth: int = 0,
   121	    ) -> None:
   122	        is_nested = depth != 0
   123	        src = self.src
   124	        last_sym_start = len(out_elements)
   125	        while char_at(src, self.pos):
   126	            if char_at(src, self.pos) == '"':
   127	                self.pos += 1
   128	                last_sym_start = len(out_elements)
   129	                while char_at(src, self.pos) != '"':
   130	                    self.check_duration()

   132	                    out_elements.append(
   133	                        InternalRuleDef(InternalRuleType.CHAR, [value])
   134	                    )
   135	                    # Adjusting pos by the length of parsed characters
   136	                    self.pos += inc_pos
   137	                self.pos = parse_space(src, self.pos + 1, is_nested)
   138	            elif char_at(src, self.pos) == "[":
   139	                self.pos += 1
   140	                start_type = InternalRuleType.CHAR
   141	                if char_at(src, self.pos) == "^":
   142	                    self.pos += 1
   143	                    start_type = InternalRuleType.CHAR_NOT
   144	                last_sym_start = len(out_elements)
   145	                while char_at(src, self.pos) != "]":
   146	                    self.check_duration()
   147	                    type_ = (
   148	                        InternalRuleType.CHAR_ALT
   149	                        if last_sym_start < len(out_elements)
   150	                        else start_type
   151	                    )
   152	                    startchar_value, inc_pos = parse_char(src, self.pos)
   153	                    self.pos += inc_pos
   154	                    if type_ in (InternalRuleType.CHAR, InternalRuleType.CHAR_NOT):
   155	                        out_elements.append(
   156	                            InternalRuleDef(type_, [startchar_value])
   157	                        )
   158	                    else:
   159	                        out_elements.append(InternalRuleDef(type_, startchar_value))
   160	
   161	                    if (
   162	                        char_at(src, self.pos) == "-"
   163	                        and char_at(src, self.pos + 1) != "]"
   164	                    ):
   165	                        self.pos += 1
   166	                        endchar_value, inc_pos = parse_char(src, self.pos)
   167	                        out_elements.append(
   168	                            InternalRuleDef(
   169	                                InternalRuleType.CHAR_RNG_UPPER, endchar_value
   170	                            )
   171	                        )

   173	                self.pos = parse_space(src, self.pos + 1, is_nested)
   174	            elif is_word_char(char_at(src, self.pos)):
   175	                name = parse_name(src, self.pos)
   176	                ref_rule_id = self.get_symbol_id(name, len(name))
   177	                self.pos += len(name)
   178	                self.pos = parse_space(src, self.pos, is_nested)
   179	
   180	                last_sym_start = len(out_elements)
   181	                out_elements.append(
   182	                    InternalRuleDef(InternalRuleType.RULE_REF, ref_rule_id)
   183	                )
   184	            elif char_at(src, self.pos) == "(":

   186	                sub_rule_id = self.generate_symbol_id(rule_name)
   187	                self.parse_alternates(rule_name, sub_rule_id, depth + 1)
   188	                last_sym_start = len(out_elements)
   189	                out_elements.append(
   190	                    InternalRuleDef(InternalRuleType.RULE_REF, sub_rule_id)
   191	                )
   192	                if char_at(src, self.pos) != ")":
   193	                    raise GrammarParseError(
   194	                        src, self.pos, f"Expecting ')' at {self.pos}"
   195	                    )
   196	                self.pos = parse_space(src, self.pos + 1, is_nested)
   197	            elif char_at(src, self.pos) in ("*", "+", "?"):
   198	                if last_sym_start == len(out_elements):
   199	                    raise GrammarParseError(
   200	                        src,
   201	                        self.pos,
   202	                        f"Expecting preceding item to */+/? at {self.pos}",
   203	                    )
   204	                sub_rule_id = self.generate_symbol_id(rule_name)
   205	                sub_rule = out_elements[last_sym_start:]
   206	                if char_at(src, self.pos) in ("*", "+"):
   207	                    sub_rule.append(
   208	                        InternalRuleDef(InternalRuleType.RULE_REF, sub_rule_id)
   209	                    )
   210	                sub_rule.append(InternalRuleDef(InternalRuleType.ALT))
   211	                if char_at(src, self.pos) == "+":
   212	                    sub_rule.extend(out_elements[last_sym_start:])
   213	                sub_rule.append(InternalRuleDef(InternalRuleType.END))
   214	                self.add_rule(sub_rule_id, sub_rule)

     1	from typing import List
     2	
     3	MAXIMUM_NUMBER_OF_ERROR_LINES_TO_SHOW = 3
     4	
     5	
     6	def build_error_position(src: str, pos: int) -> List[str]:
     7	    if src == "":
     8	        return ["No input provided"]
     9	
    10	    lines = src.split("\n")
    11	
    12	    line_idx = 0
    13	    # `lines[line_idx]` is falsy both past the end of the list and for empty lines,
    14	    # which is what stops the walk in the reference implementation.
    15	    while line_idx < len(lines) and lines[line_idx] and pos > len(lines[line_idx]) - 1:
    16	        pos -= len(lines[line_idx])
    17	        line_idx += 1
    18	
    19	    lines_to_show: List[str] = []
    20	    start = max(0, line_idx - (MAXIMUM_NUMBER_OF_ERROR_LINES_TO_SHOW - 1))
    21	    for i in range(start, line_idx + 1):
    22	        lines_to_show.append(lines[i] if i < len(lines) else "")
    23	
    24	    return [*lines_to_show, " " * pos + "^"]
