     3	The original pairs a ``Map`` keyed by a derived key with a ``Set`` keyed by
     4	object identity. Rules and pointers define value equality (and are therefore
     5	unhashable), so identity here is tracked with ``id()``; the dict holds a strong
     6	reference to every member, which keeps those ids stable.

     9	from __future__ import annotations
    10	
    11	from typing import Callable, Dict, Generic, Iterator, Optional, TypeVar
    12	
    13	T = TypeVar('T')
    14	K = TypeVar('K')
    15	
    16	
    17	class GenericSet(Generic[T, K]):
    18	    def __init__(self, get_key: Callable[[T], K]) -> None:
    19	        self.get_key = get_key
    20	        self._keys: Dict[K, T] = {}
    21	        self._set: Dict[int, T] = {}
    22	
    23	    def add(self, el: T) -> None:
    24	        key = self.get_key(el)
    25	        if key not in self._keys:
    26	            self._keys[key] = el
    27	            self._set[id(el)] = el
    28	
    29	    def delete(self, el: T) -> None:
    30	        key = self.get_key(el)
    31	        ref = self._keys.get(key)
    32	        if ref is None:

    34	        del self._keys[key]
    35	        self._set.pop(id(ref), None)
    36	
    37	    def has(self, el: T) -> bool:
    38	        return id(el) in self._set
    39	
    40	    def get(self, el: T) -> Optional[T]:
    41	        return self._keys.get(self.get_key(el))
    42	
    43	    def __iter__(self) -> Iterator[T]:

   109	    def _parse(self, current_pointers: Pointers, code_point: int) -> Pointers:
   110	        for rule, graph_pointers in self._iterate_over_pointers(current_pointers):
   111	            if is_rule_char(rule):
   112	                valid = False
   113	                for possible_code_point in rule.value:
   114	                    if valid:
   115	                        break
   116	                    if is_range(possible_code_point):
   117	                        valid = is_point_in_range(code_point, possible_code_point)
   118	                    else:
   119	                        valid = code_point == possible_code_point
   120	                self._set_valid(graph_pointers, valid)
   121	            elif is_rule_char_excluded(rule):
   122	                valid = True
   123	                for possible_code_point in rule.value:
   124	                    if not valid:
   125	                        break
   126	                    if is_range(possible_code_point):
   127	                        valid = not is_point_in_range(code_point, possible_code_point)
   128	                    else:
   129	                        valid = code_point != possible_code_point
   130	                self._set_valid(graph_pointers, valid)
   131	            elif not is_rule_end(rule):
   132	                raise ValueError(f'Unsupported rule: {rule!r}')
   133	
   134	        # a pointer's id is the sum of its node's id and its parent's id chain.
   135	        # if two pointers share the same id, it means they point to the same node and have
   136	        # identical parent chains. for the purposes of walking the graph, we only need to
   137	        # keep one of them.
   138	        next_pointers = make_pointers()
   139	        for current_pointer in current_pointers:
   140	            for unresolved_next_pointer in current_pointer.fetch_next():
   141	                for resolved_next_pointer in self._resolve_pointer(unresolved_next_pointer):
   142	                    next_pointers.add(resolved_next_pointer)
   143	        return next_pointers

   145	    def _resolve_pointer(self, unresolved_pointer: GraphPointer) -> Iterator[GraphPointer]:
   146	        for resolved_pointer in unresolved_pointer.resolve():
   147	            if is_rule_ref(resolved_pointer.node.rule):
   148	                raise ValueError(
   149	                    'Encountered a reference rule when building pointers to the graph'
   150	                )
   151	            if is_rule_end(resolved_pointer.node.rule) and resolved_pointer.parent is not None:
   152	                raise ValueError(
   153	                    'Encountered an ending rule with a parent when building pointers to '
   154	                    'the graph'
   155	                )
   156	            yield resolved_pointer

   159	        pointers = _pointers if _pointers is not None else self._get_initial_pointers()
   160	        code_points = get_input_as_code_points(src)
   161	        for code_point_pos, code_point in enumerate(code_points):
   162	            pointers = self._parse(pointers, code_point)

   164	                raise InputParseError(code_points, code_point_pos, self._previous_code_points)

    22	class GraphPointer:
    23	    __slots__ = ('node', 'parent', '_valid', 'id')
    24	
    25	    def __init__(self, node: GraphNode, parent: Optional['GraphPointer'] = None) -> None:
    26	        if node is None:
    27	            raise ValueError('Node is undefined')
    28	        self.node = node
    29	        self.parent = parent
    30	        self._valid: Optional[bool] = None
    31	        self.id = f'{parent.id}-{node.id}' if parent is not None else node.id
    32	
    33	    def resolve(self, resolved: bool = False) -> Iterator['GraphPointer']:
    34	        """
    35	        1. If the current node is an end node, and the pointer has a parent, return the
    36	           parent's `fetch_next`; else return nothing.
    37	        2. If the current node is a rule ref, yield the referenced nodes, _unless_
    38	           resolved is true, in which case it returns next.
    39	        3. If the current node is a char or range, we go to the next node. If none
    40	           exists, throw an error.
    41	
    42	        The reference implementation recurses. Here the walk uses an explicit stack:
    43	        a pointer's parent chain grows with every repetition of a self-referential
    44	        rule, so recursing would exhaust Python's much smaller stack on long inputs.
    45	        The stack is LIFO with children pushed in reverse, which reproduces the
    46	        original's depth-first, left-to-right yield order.
    47	        """
    48	        stack: List[Tuple['GraphPointer', bool]] = [(self, resolved)]
    49	        while stack:
    50	            pointer, is_resolved = stack.pop()
    51	            if is_graph_pointer_rule_ref(pointer):
    52	                if is_resolved:
    53	                    if pointer.node.next is None:
    54	                        raise ValueError(f'No next node: {pointer.node!r}')
    55	                    stack.append((GraphPointer(pointer.node.next, pointer.parent), False))
    56	                else:
    57	                    stack.extend(
    58	                        (GraphPointer(node, pointer), False)
    59	                        for node in reversed(pointer.node.rule.nodes)
    60	                    )
    61	            elif is_graph_pointer_rule_end(pointer):
    62	                if pointer.parent is None:
    63	                    yield pointer
    64	                else:
    65	                    stack.append((pointer.parent, True))
    66	            elif is_graph_pointer_rule_char(pointer) or is_graph_pointer_rule_char_exclude(
    67	                pointer
    68	            ):
    69	                yield pointer
    70	            else:
    71	                raise ValueError(f'Unknown rule: {pointer.node.rule!r}')
    72	
    73	    def fetch_next(self) -> Iterator['GraphPointer']:
    74	        # Walks up the parent chain iteratively, for the same reason `resolve` does.
    75	        pointer: Optional['GraphPointer'] = self
    76	        while pointer is not None:
    77	            # if this pointer is invalid, then we don't return any new pointers
    78	            if pointer._valid is False:
    79	                return
    80	
    81	            # if this pointer is an end node, we return the parent's next node. If no
    82	            # parent exists, we return nothing, since it's the end of the line.
    83	            if is_rule_end(pointer.node.rule):
    84	                pointer = pointer.parent
    85	            else:
    86	                if pointer.node.next is None:
    87	                    raise ValueError(f'No next node: {pointer.node!r}')
    88	                yield from GraphPointer(pointer.node.next, pointer.parent).resolve()
    89	                return

    22	def _make_char_rule(rule_def: InternalRuleDef):
    23	    value = list(rule_def.value)
    24	    return RuleCharExclude(value) if is_rule_def_char_not(rule_def) else RuleChar(value)
    25	
    26	
    27	def build_rule_stack(linear_rules: List[InternalRuleDef]) -> List[List[UnresolvedRule]]:
    28	    paths: List[UnresolvedRule] = []
    29	
    30	    stack: List[List[UnresolvedRule]] = []
    31	
    32	    idx = 0
    33	    while idx < len(linear_rules):
    34	        rule_def = linear_rules[idx]
    35	        if is_rule_def_char(rule_def) or is_rule_def_char_not(rule_def):
    36	            # this could be a single char, or a range, or a sequence of alts; we don't
    37	            # know until we step through it.
    38	            char_rule = _make_char_rule(rule_def)
    39	            idx += 1
    40	            rule = linear_rules[idx] if idx < len(linear_rules) else None
    41	            while idx < len(linear_rules) and (
    42	                is_rule_def_char_rng_upper(rule) or is_rule_def_char_alt(rule)
    43	            ):
    44	                if is_rule_def_char_rng_upper(rule):
    45	                    # previous rule value should be a number
    46	                    prev_value = char_rule.value.pop() if char_rule.value else None
    47	                    if is_range(prev_value):
    48	                        raise ValueError(
    49	                            'Unexpected range, expected a number but got an array: '
    50	                            f'{prev_value}'
    51	                        )
    52	                    if prev_value is None:
    53	                        raise ValueError('Unexpected undefined value')
    54	                    char_rule.value.append([prev_value, rule.value])
    55	                if is_rule_def_char_alt(rule):
    56	                    char_rule.value.append(rule.value)
    57	                idx += 1
    58	                rule = linear_rules[idx] if idx < len(linear_rules) else None
    59	            paths.append(char_rule)
    60	        else:
    61	            if is_rule_def_alt(rule_def):
    62	                if not paths:
    63	                    raise ValueError('Encountered alt without anything before it')
    64	                paths.append(RuleEnd())
    65	                stack.append(paths)
    66	                paths = []
    67	            elif is_rule_def_end(rule_def):
    68	                paths.append(RuleEnd())
    69	            elif is_rule_def_ref(rule_def):
    70	                paths.append(RuleRef(rule_def.value))
    71	            elif is_rule_def_char_alt(rule_def):
    72	                raise ValueError(
    73	                    'Encountered char alt, should be handled by above block: '
    74	                    f'{rule_def.type}'
    75	                )
    76	            else:
    77	                raise ValueError(f'Unsupported rule type: {rule_def.type}')
    78	            idx += 1
    79	
    80	    if not paths or not is_rule_end(paths[-1]):
    81	        paths.append(RuleEnd())
    82	
    83	    stack.append(paths)
    84	
    85	    return stack
