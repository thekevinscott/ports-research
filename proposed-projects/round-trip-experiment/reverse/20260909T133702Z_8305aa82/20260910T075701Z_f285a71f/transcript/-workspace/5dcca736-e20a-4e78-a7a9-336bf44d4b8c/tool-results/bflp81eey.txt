    25	    stack: List[List] = []
    26	
    27	    idx = 0
    28	    while idx < len(linear_rules):
    29	        rule_def = linear_rules[idx]
    30	        if is_rule_def_char(rule_def) or is_rule_def_char_not(rule_def):
    31	            # this could be a single char, or a range, or a sequence of alts;
    32	            # we don't know until we step through it.
    33	            char_rule = _make_char_rule(rule_def)

    45	                            f"{prev_value!r}"
    46	                        )

    50	                if is_rule_def_char_alt(rule):
    51	                    char_rule.value.append(rule.value)
    52	                idx += 1

    63	                paths.append(RuleEnd())

    69	                    f"{rule_def.type}"

    71	            else:
    72	                raise Exception(f"Unsupported rule type: {rule_def.type}")

     2	
     3	Mirrors the reference `GenericSet`: adding an element whose key is already
     4	present is a no-op, so the first element added for a key wins and keeps its
     5	position in iteration order.
     6	"""
     7	
     8	from typing import Callable, Dict, Generic, Iterator, TypeVar
     9	
    10	T = TypeVar("T")
    11	K = TypeVar("K")
    12	
    13	
    14	class GenericSet(Generic[T, K]):
    15	    def __init__(self, get_key: Callable[[T], K]):
    16	        self.get_key = get_key
    17	        self._keys: Dict[K, T] = {}
    18	
    19	    def add(self, el: T) -> None:
    20	        key = self.get_key(el)
    21	        if key not in self._keys:
    22	            self._keys[key] = el
    23	
    24	    def delete(self, el: T) -> None:
    25	        key = self.get_key(el)
    26	        if key not in self._keys:
    27	            raise Exception("Could not get ref")
    28	        del self._keys[key]
    29	
    30	    def has(self, el: T) -> bool:
    31	        key = self.get_key(el)
    32	        return key in self._keys and self._keys[key] is el
    33

    35	        return self._keys.get(self.get_key(el))

   103	    def _parse(self, current_pointers: GenericSet, code_point: int) -> GenericSet:
   104	        for rule, graph_pointers in self._iterate_over_pointers(current_pointers):
   105	            if is_rule_char(rule):
   106	                valid = False
   107	                for possible_code_point in rule.value:
   108	                    if valid:
   109	                        break
   110	                    if is_range(possible_code_point):
   111	                        valid = is_point_in_range(code_point, possible_code_point)
   112	                    else:
   113	                        valid = code_point == possible_code_point
   114	                self._set_valid(graph_pointers, valid)
   115	            elif is_rule_char_excluded(rule):
   116	                valid = True
   117	                for possible_code_point in rule.value:
   118	                    if not valid:
   119	                        break
   120	                    if is_range(possible_code_point):
   121	                        valid = not is_point_in_range(code_point, possible_code_point)
   122	                    else:
   123	                        valid = code_point != possible_code_point
   124	                self._set_valid(graph_pointers, valid)
   125	            elif not is_rule_end(rule):
   126	                raise Exception(f"Unsupported rule: {rule!r}")
   127	
   128	        # a pointer's id is the sum of its node's id and its parent's id chain.
   129	        # if two pointers share the same id, it means they point to the same node and have
   130	        # identical parent chains.
   131	        # for the purposes of walking the graph, we only need to keep one of them.
   132	        next_pointers = make_pointers()
   133	        for current_pointer in current_pointers:
   134	            for unresolved_next_pointer in current_pointer.fetch_next():
   135	                for resolved_next_pointer in self._resolve_pointer(
   136	                    unresolved_next_pointer
   137	                ):
   138	                    next_pointers.add(resolved_next_pointer)

   148	                raise Exception(

   150	                )

    13	    def __init__(self, node, parent=None):
    14	        if node is None:
    15	            raise Exception("Node is undefined")
    16	        self.node = node
    17	        self.parent = parent
    18	        self.id = f"{parent.id}-{node.id}" if parent is not None else node.id
    19	        self._valid = None
    20	
    21	    def resolve(self, resolved=False):
    22	        """
    23	        1. If the current node is an end node, and the pointer has a parent, return the parent's `fetch_next`; else return nothing.
    24	        2. If the current node is a rule ref, yield the referenced nodes, _unless_ resolved is true, in which case it returns next.
    25	        3. If the current node is a char or range, we go to the next node. If none exists, throw an error.
    26	        """
    27	        if is_graph_pointer_rule_ref(self):
    28	            if resolved:
    29	                if self.node.next is None:
    30	                    raise Exception(f"No next node: {self.node!r}")
    31	                yield from GraphPointer(self.node.next, self.parent).resolve()
    32	            else:
    33	                for node in self.node.rule.nodes:
    34	                    yield from GraphPointer(node, self).resolve()
    35	        elif is_graph_pointer_rule_end(self):
    36	            if self.parent is None:
    37	                yield self
    38	            else:
    39	                yield from self.parent.resolve(True)
    40	        elif is_graph_pointer_rule_char(self) or is_graph_pointer_rule_char_exclude(
    41	            self
    42	        ):
    43	            yield self
    44	        else:
    45	            raise Exception(f"Unknown rule: {self.node.rule!r}")
    46	
    47	    def fetch_next(self):
    48	        # if this pointer is invalid, then we don't return any new pointers
    49	        if self._valid is False:
    50	            return
    51	
    52	        # if this pointer is an end node, we return the parent's next node. If no parent exists,
    53	        # we return nothing, since it's the end of the line.
    54	        if is_rule_end(self.node.rule):
    55	            if self.parent is not None:
    56	                yield from self.parent.fetch_next()
    57	        else:
    58	            if self.node.next is None:
    59	                raise Exception(f"No next node: {self.node!r}")
    60	            pointer = GraphPointer(self.node.next, self.parent)
    61	            yield from pointer.resolve()

    10	def print_graph_node(node, pointers=None, show_position=False, colorize=None):
    11	    col = colorize
    12	    rule = node.rule
    13	
    14	    parts = []
    15	    if show_position:
    16	        parts.extend(
    17	            [
    18	                col("{", Color.BLUE),
    19	                col(node.id, Color.GRAY),
    20	                col("}", Color.BLUE),
    21	            ]
    22	        )
    23	    if is_rule_char(rule):
    24	        rendered = []
    25	        for v in rule.value:
    26	            if is_range(v):
    27	                rendered.append(",".join(col(chr(val), Color.YELLOW) for val in v))
    28	            else:
    29	                rendered.append(_get_char(v))
    30	        parts.extend(
    31	            [
    32	                col("[", Color.GRAY),
    33	                col("".join(rendered), Color.YELLOW),
    34	                col("]", Color.GRAY),
    35	            ]
    36	        )
    37	    elif is_rule_ref(rule):
    38	        parts.append(
    39	            col("Ref(", Color.GRAY)
    40	            + col(f"{rule.value}", Color.GREEN)
    41	            + col(")", Color.GRAY)
    42	        )
    43	    else:
    44	        parts.append(col(rule.type, Color.YELLOW))
    45	
    46	    if pointers:
    47	        for pointer in pointers:
    48	            pointer_parts = []
    49	            if pointer.node is node:
    50	                pointer_parts.append(pointer.print(colorize=col))
    51	            if pointer_parts:
    52	                parts.append(col("[", Color.GRAY))
    53	                parts.extend(pointer_parts)
    54	                parts.append(col("]", Color.GRAY))
    55	
    56	    rendered_parts = ["".join(parts)]
    57	    if node.next is not None:
    58	        rendered_parts.append(
    59	            node.next.print(
    60	                pointers=pointers, colorize=col, show_position=show_position
    61	            )
    62	        )
    63	    return col("-> ", Color.GRAY).join(part for part in rendered_parts if part)

    66	def _get_char(char_code: int) -> str:
    67	    char = chr(char_code)
    68	    if char == "\n":
    69	        return "\\n"
    70	    return char
