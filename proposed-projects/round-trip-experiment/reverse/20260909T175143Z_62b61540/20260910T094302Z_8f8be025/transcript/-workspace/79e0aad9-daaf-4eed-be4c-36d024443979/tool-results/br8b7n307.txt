    16	
    17	export type GraphPointerKey = string;
    18	
    19	export class GraphPointer<T extends UnresolvedRule = UnresolvedRule> {
    20	  node: GraphNode<T>;
    21	  parent?: GraphPointer;
    22	  id: GraphPointerKey;
    23	  #valid?: boolean;
    24	
    25	  constructor(node: GraphNode<T>, parent?: GraphPointer) {
    26	    if (node === undefined) {
    27	      throw new Error('Node is undefined');
    28	    }
    29	    this.node = node;
    30	    this.parent = parent;
    31	    this.id = parent ? `${parent.id}-${node.id}` : node.id;
    32	  }
    33	
    34	  get rule(): T {
    35	    return this.node.rule;
    36	  }
    37	
    38	  get valid(): boolean | undefined {
    39	    return this.#valid;
    40	  }
    41	
    42	  set valid(valid: boolean | undefined) {
    43	    this.#valid = valid;
    44	  }
    45	
    46	  print(opts: PrintOpts): string {
    47	    return printGraphPointer(this as GraphPointer)(opts);
    48	  }
    49	
    50	  /**
    51	   * Resolve this pointer into pointers that point at concrete (non-reference) rules.
    52	   *
    53	   * @param resolved whether the pointer has already been resolved
    54	   */
    55	  *resolve(resolved = false): Generator<ResolvedGraphPointer> {
    56	    const pointer = this as GraphPointer;
    57	    if (isGraphPointerRuleRef(pointer)) {
    58	      if (resolved) {
    59	        if (!pointer.node.next) {
    60	          throw new Error(`No next node: ${pointer.node.id}`);
    61	        }
    62	        yield* new GraphPointer(pointer.node.next, pointer.parent).resolve();
    63	      } else {
    64	        for (const node of pointer.rule.nodes) {
    65	          yield* new GraphPointer(node, pointer).resolve();
    66	        }
    67	      }
    68	    } else if (isGraphPointerRuleEnd(pointer)) {
    69	      if (!pointer.parent) {
    70	        yield pointer as GraphPointer<ResolvedRule>;
    71	      } else {
    72	        yield* pointer.parent.resolve(true);
    73	      }
    74	    } else if (isGraphPointerRuleChar(pointer) || isGraphPointerRuleCharExclude(pointer)) {
    75	      yield pointer as GraphPointer<ResolvedRule>;
    76	    } else {
    77	      throw new Error(`Unknown rule: ${pointer.node.rule}`);
    78	    }
    79	  }
    80	
    81	  /**
    82	   * Fetch the pointers that follow this one, if this pointer was valid.
    83	   */
    84	  *fetchNext(): Generator<GraphPointer> {
    85	    // If this pointer is invalid, then we don't return any new pointers
    86	    if (!this.valid) {
    87	      return;
    88	    }
    89	
    90	    // If this pointer is an end node, we return the parent's next node.
    91	    // If no parent exists, we return nothing, since it's the end of the line.
    92	    if (isRuleEnd(this.node.rule)) {
    93	      if (this.parent) {
    94	        yield* this.parent.fetchNext();
    95	      }
    96	    } else {
    97	      if (!this.node.next) {
    98	        throw new Error(`No next node: ${this.node.id}`);
    99	      }
   100	      const pointer = new GraphPointer(this.node.next, this.parent);
   101	      yield* pointer.resolve();
   102	    }
   103	  }
   104	}

    22	
    23	export type RootNode = Map<number, GraphNode>;
    24	
    25	export class Graph {
    26	  #roots = new Map<number, RootNode>();
    27	  #rootNode: RootNode;
    28	  grammar: string;
    29	  previousCodePoints: number[] = [];
    30	
    31	  constructor(grammar: string, stackedRules: UnresolvedRule[][][], rootId: number) {
    32	    this.grammar = grammar;
    33	    const ruleRefs: RuleRef[] = [];
    34	    const uniqueRules = new Map<string, UnresolvedRule>();
    35	
    36	    for (let stackId = 0; stackId < stackedRules.length; stackId++) {
    37	      const stack = stackedRules[stackId];
    38	      const nodes: RootNode = new Map();
    39	      for (let pathId = 0; pathId < stack.length; pathId++) {
    40	        const path = stack[pathId];
    41	        let node: GraphNode | undefined;
    42	        for (let stepId = path.length - 1; stepId >= 0; stepId--) {
    43	          const nextNode = node;
    44	          const rule = path[stepId];
    45	          uniqueRules.set(getSerializedRuleKey(rule), rule);
    46	          if (isRuleRef(rule)) {
    47	            ruleRefs.push(rule);
    48	          }
    49	          // rules coming in may be identical but have different references.
    50	          // here, we ensure we always use the same reference for an identical rule.
    51	          // this makes future comparisons easier.
    52	          const uniqueRule = uniqueRules.get(getSerializedRuleKey(rule));
    53	          if (uniqueRule === undefined) {
    54	            throw new Error('Could not get unique rule');
    55	          }
    56	          node = new GraphNode(uniqueRule, { stackId, pathId, stepId }, nextNode);
    57	        }
    58	
    59	        if (node === undefined) {
    60	          throw new Error('Could not get node');
    61	        }
    62	        nodes.set(pathId, node);
    63	      }
    64	      this.#roots.set(stackId, nodes);
    65	    }
    66	
    67	    const rootNode = this.#roots.get(rootId);
    68	    if (rootNode === undefined) {
    69	      throw new Error(`Root node not found for value: ${rootId}`);
    70	    }
    71	    this.#rootNode = rootNode;
    72	
    73	    for (const ruleRef of ruleRefs) {
    74	      ruleRef.nodes = new Set(this.#getRootNode(ruleRef.value).values());
    75	    }
    76	  }
    77	
    78	  #getRootNode(value: number): RootNode {
    79	    const rootNode = this.#roots.get(value);
    80	    if (rootNode === undefined) {
    81	      throw new Error(`Root node not found for value: ${value}`);
    82	    }
    83	    return rootNode;
    84	  }
    85	
    86	  #getInitialPointers(): Pointers {
    87	    const pointers = new Pointers();
    88	
    89	    const rootNode = this.#rootNode;
    90	    if (rootNode === undefined) {
    91	      throw new Error('Root node is not defined');
    92	    }
    93	
    94	    for (const [node, parent] of this.#fetchNodesForRootNode(rootNode)) {
    95	      const pointer = new GraphPointer(node, parent);
    96	      for (const resolvedPointer of this.#resolvePointer(pointer)) {
    97	        pointers.add(resolvedPointer);
    98	      }
    99	    }
   100	    return pointers;
   101	  }

   103	  #setValid(pointers: GraphPointer[], valid: boolean): void {
   104	    for (const pointer of pointers) {
   105	      pointer.valid = valid;
   106	    }
   107	  }

   109	  #parse(currentPointers: Pointers, codePoint: number): Pointers {
   110	    for (const [rule, graphPointers] of this.#iterateOverPointers(currentPointers)) {
   111	      if (isRuleChar(rule)) {
   112	        let valid = false;
   113	        for (const possibleCodePoint of rule.value) {
   114	          if (valid === true) {
   115	            // already valid, no need to check further
   116	          } else if (isRange(possibleCodePoint)) {
   117	            if (isPointInRange(codePoint, possibleCodePoint)) {
   118	              valid = true;
   119	            }
   120	          } else if (codePoint === possibleCodePoint) {
   121	            valid = true;
   122	          }
   123	        }
   124	        this.#setValid(graphPointers, valid);
   125	      } else if (isRuleCharExclude(rule)) {
   126	        let valid = true;
   127	        for (const possibleCodePoint of rule.value) {
   128	          if (valid === false) {
   129	            // already invalid, no need to check further
   130	          } else if (isRange(possibleCodePoint)) {
   131	            if (isPointInRange(codePoint, possibleCodePoint)) {
   132	              valid = false;
   133	            }
   134	          } else if (codePoint === possibleCodePoint) {
   135	            valid = false;
   136	          }
   137	        }
   138	        this.#setValid(graphPointers, valid);
   139	      } else if (!isRuleEnd(rule)) {
   140	        throw new Error(`Unsupported rule: ${rule}`);
   141	      }
   142	    }
   143	
   144	    // a pointer's id is the sum of its node's id and its parent's id chain.
   145	    // if two pointers share the same id, it means they point to the same node and have identical
   146	    // parent chains. for the purposes of walking the graph, we only need to keep one of them.
   147	    const nextPointers = new Pointers();
   148	    for (const currentPointer of currentPointers) {
   149	      for (const unresolvedNextPointer of currentPointer.fetchNext()) {
   150	        for (const resolvedNextPointer of this.#resolvePointer(unresolvedNextPointer)) {
   151	          nextPointers.add(resolvedNextPointer);
   152	        }
   153	      }
   154	    }
   155	    return nextPointers;
   156	  }
   157	
   158	  *#resolvePointer(unresolvedPointer: GraphPointer): Generator<ResolvedGraphPointer> {
   159	    for (const resolvedPointer of unresolvedPointer.resolve()) {
   160	      if (isRuleRef(resolvedPointer.node.rule)) {
   161	        throw new Error('Encountered a reference rule when building pointers to the graph');
   162	      }
   163	      if (isRuleEnd(resolvedPointer.node.rule) && resolvedPointer.parent !== undefined) {
   164	        throw new Error(
   165	          'Encountered an ending rule with a parent when building pointers to the graph',
   166	        );
   167	      }
   168	      yield resolvedPointer;
   169	    }
   170	  }

   172	  add(src: ValidInput, pointers?: Pointers): Pointers {
   173	    if (typeof src !== 'string') {
   174	      throw new Error('src must be a string in graph.add');
   175	    }
   176	    let currentPointers = pointers ?? this.#getInitialPointers();
   177	
   178	    const codePoints = getInputAsCodePoints(src);
   179	
   180	    for (let codePointPos = 0; codePointPos < codePoints.length; codePointPos++) {
   181	      const codePoint = codePoints[codePointPos];
   182	      currentPointers = this.#parse(currentPointers, codePoint);
   183	      if (currentPointers.size === 0) {
   184	        throw new InputParseError(codePoints, codePointPos, this.previousCodePoints);
   185	      }
   186	    }
   187	    this.previousCodePoints.push(...codePoints);
   188	    return currentPointers;
   189	  }
   190	
   191	  // generator that yields either the node, or if a reference rule, the referenced node.
   192	  // we need this function, as distinct from leveraging the logic in GraphPointer,
   193	  // because that needs a rule ref with already defined nodes; this function is used to _set_ those.
   194	  *#fetchNodesForRootNode(
   195	    rootNodes: RootNode,
   196	    parent?: GraphPointer,
   197	  ): Generator<[GraphNode, GraphPointer | undefined]> {
   198	    for (const node of rootNodes.values()) {
   199	      const { rule } = node;
   200	      if (isRuleRef(rule)) {
   201	        yield* this.#fetchNodesForRootNode(
   202	          this.#getRootNode(rule.value),
   203	          new GraphPointer(node, parent),
   204	        );
   205	      } else {
   206	        yield [node, parent];
   207	      }
   208	    }
   209	  }

   211	  print(pointers?: Pointers, colors = false): string {
   212	    const graphView: string[] = [];
   213	    for (const rootNode of this.#roots.values()) {
   214	      for (const node of rootNode.values()) {
   215	        graphView.push(node.print({
   216	          pointers: pointers ?? new Pointers(),
   217	          showPosition: true,
   218	          colorize: colors ? colorize : (value) => `${value}`,
   219	        }));
   220	      }
   221	    }
   222	
   223	    return graphView.join('\n');
   224	  }
   225	
   226	  *#iterateOverPointers(
   227	    pointers: Pointers,
   228	  ): Generator<[UnresolvedRule, GraphPointer[]]> {
   229	    const seenRules = new Map<UnresolvedRule, GraphPointer[]>();
   230	    for (const pointer of pointers) {
   231	      const { rule } = pointer;
   232	      if (isRuleRef(rule)) {
   233	        throw new Error('Encountered a reference rule in the graph');
   234	      }
   235	
   236	      let seenRule = seenRules.get(rule);
   237	      if (seenRule === undefined) {
   238	        seenRule = [pointer];
   239	        seenRules.set(rule, seenRule);
   240	      }
   241	      seenRule.push(pointer);
   242	    }
   243	
   244	    yield* seenRules.entries();
   245	  }

    36	export const buildRuleStack = (linearRules: InternalRuleDef[]): UnresolvedRule[][] => {
    37	  let paths: UnresolvedRule[] = [];
    38	  const stack: UnresolvedRule[][] = [];
    39	  let idx = 0;
    40	
    41	  while (idx < linearRules.length) {
    42	    const ruleDef = linearRules[idx];
    43	    if (isRuleDefChar(ruleDef) || isRuleDefCharNot(ruleDef)) {
    44	      // this could be a single char, or a range, or a sequence of alts;
    45	      // we don't know until we step through it.
    46	      const charRule = makeCharRule(ruleDef);
    47	      idx += 1;
    48	      let rule: InternalRuleDef | undefined = linearRules[idx];
    49	      while (idx < linearRules.length && (isRuleDefCharRngUpper(rule) || isRuleDefCharAlt(rule))) {
    50	        if (isRuleDefCharRngUpper(rule)) {
    51	          // previous rule value should be a number
    52	          const prevValue = charRule.value.pop();

    54	            throw new Error(
    55	              `Unexpected range, expected a number but got an array: ${JSON.stringify(prevValue)}`,
    56	            );
    57	          }
    58	          if (prevValue === undefined) {
    59	            throw new Error('Unexpected undefined value');
    60	          }

    62	          charRule.value.push([prevValue, rule.value] as Range);
    63	        }
    64	        if (isRuleDefCharAlt(rule)) {
    65	          charRule.value.push(rule.value);
    66	        }
    67	        idx += 1;
    68	        rule = linearRules[idx];
    69	      }

    71	    } else {

    81	      } else if (isRuleDefRef(ruleDef)) {
    82	        paths.push(new RuleRef(ruleDef.value));

    12	  if (pos >= src.length) {
    13	    throw new GrammarParseError(
    14	      src,

   157	        while (src[this.pos] !== '"') {
   158	          this.checkDuration();
   159	          const [value, incPos] = parseChar(src, this.pos);
   160	          outElements.push({ type: InternalRuleType.CHAR, value: [value] });
   161	          this.pos += incPos;
   162	        }
   163	        this.pos = parseSpace(src, this.pos + 1, isNested);
   164	      } else if (src[this.pos] === '[') {
   165	        this.pos += 1;
   166	        let startType = InternalRuleType.CHAR;
   167	        if (src[this.pos] === '^') {
   168	          this.pos += 1;
   169	          startType = InternalRuleType.CHAR_NOT;
   170	        }
   171	        lastSymStart = outElements.length;
   172	        while (src[this.pos] !== ']') {
   173	          this.checkDuration();
   174	          const type = lastSymStart < outElements.length ? InternalRuleType.CHAR_ALT : startType;
   175	          const [startCharValue, incPos] = parseChar(src, this.pos);
   176	          this.pos += incPos;
   177	          outElements.push(getOutElements(type, startCharValue));
   178	
   179	          if (src[this.pos] === '-' && src[this.pos + 1] !== ']') {
   180	            this.pos += 1;
   181	            const [endCharValue, endIncPos] = parseChar(src, this.pos);
   182	            outElements.push({
   183	              type: InternalRuleType.CHAR_RNG_UPPER,
   184	              value: endCharValue,
   185	            });
   186	            this.pos += endIncPos;
   187	          }
   188	        }
   189	        this.pos = parseSpace(src, this.pos + 1, isNested);
   190	      } else if (isWordChar(src[this.pos])) {
   191	        const name = parseName(src, this.pos);
   192	        const refRuleId = this.getSymbolId(name, name.length);
   193	        this.pos += name.length;
   194	        this.pos = parseSpace(src, this.pos, isNested);
   195	
   196	        lastSymStart = outElements.length;
   197	        outElements.push({ type: InternalRuleType.RULE_REF, value: refRuleId });
   198	      } else if (src[this.pos] === '(') {
   199	        this.pos = parseSpace(src, this.pos + 1, true);
   200	        const subRuleId = this.generateSymbolId(ruleName);
   201	        this.parseAlternates(ruleName, subRuleId, depth + 1);
   202	        lastSymStart = outElements.length;
   203	        outElements.push({ type: InternalRuleType.RULE_REF, value: subRuleId });
   204	        if (src[this.pos] !== ')') {
   205	          throw new GrammarParseError(src, this.pos, `Expecting ')' at ${this.pos}`);
   206	        }
   207	        this.pos = parseSpace(src, this.pos + 1, isNested);
   208	      } else if (MODIFIERS.includes(src[this.pos])) {
   209	        if (lastSymStart === outElements.length) {
   210	          throw new GrammarParseError(
   211	            src,
   212	            this.pos,
   213	            `Expecting preceding item to */+/? at ${this.pos}`,
   214	          );
   215	        }
   216	        const subRuleId = this.generateSymbolId(ruleName);
   217	        const subRule = outElements.slice(lastSymStart);
   218	        if (src[this.pos] === '*' || src[this.pos] === '+') {
   219	          subRule.push({ type: InternalRuleType.RULE_REF, value: subRuleId });
   220	        }
   221	        subRule.push({ type: InternalRuleType.ALT });
   222	        if (src[this.pos] === '+') {
   223	          subRule.push(...outElements.slice(lastSymStart));
   224	        }
   225	        subRule.push({ type: InternalRuleType.END });
   226	        this.addRule(subRuleId, subRule);
   227	        outElements.splice(lastSymStart, outElements.length - lastSymStart, {
   228	          type: InternalRuleType.RULE_REF,
   229	          value: subRuleId,
   230	        });
   231	        this.pos = parseSpace(src, this.pos + 1, isNested);
   232	      } else {
   233	        break;
   234	      }
   235	    }
   236	  }
   237	
   238	  parseAlternates(ruleName: string, ruleId: number, depth = 0): void {
   239	    const { src } = this;
   240	    const rule: InternalRuleDef[] = [];
   241	    this.parseSequence(ruleName, rule, depth);
   242	    // Ensure that this.pos is within bounds before checking src[this.pos]
   243	    while (this.pos < src.length && src[this.pos] === '|') {
   244	      this.checkDuration();
   245	      rule.push({ type: InternalRuleType.ALT });
   246	      this.pos = parseSpace(src, this.pos + 1, true);
   247	      this.parseSequence(ruleName, rule, depth);
   248	    }
   249	    rule.push({ type: InternalRuleType.END });
   250	    this.addRule(ruleId, rule);
   251	  }
   252	}

     3	export const buildErrorPosition = (src: string, pos: number): string[] => {
     4	  if (src === '') {
     5	    return ['No input provided'];
     6	  }
     7	  const lines = src.split('\n');
     8	
     9	  let lineIdx = 0;
    10	  while (lines[lineIdx] && pos > lines[lineIdx].length - 1 && pos < src.length) {
    11	    pos -= lines[lineIdx].length;
    12	    lineIdx += 1;
    13	  }
    14	
    15	  const linesToShow = lines.slice(
    16	    Math.max(0, lineIdx - (MAXIMUM_NUMBER_OF_ERROR_LINES_TO_SHOW - 1)),
    17	    lineIdx + 1,
    18	  );
    19	
    20	  return [
    21	    ...linesToShow,
    22	    `${' '.repeat(Math.max(0, pos))}^`,
    23	  ];
