    32	        self._previous_code_points: List[int] = []
    33

    36	        for stack_id, stack in enumerate(stacked_rules):
    37	            if stack is None:
    38	                continue
    39	            nodes: RootNode = {}
    40	            for path_id, path in enumerate(stack):
    41	                node: Optional[GraphNode] = None
    42	                for step_id in range(len(path) - 1, -1, -1):
    43	                    next_ = node
    44	                    rule = stack[path_id][step_id]
    45	                    unique_rules.add(rule)
    46	                    if is_rule_ref(rule):
    47	                        rule_refs.append(rule)
    48	                    # rules coming in may be identical but have different references.
    49	                    # here, we ensure we always use the same reference for an identical
    50	                    # rule. this makes future comparisons easier.
    51	                    unique_rule = unique_rules.get(rule)
    52	                    if unique_rule is None:
    53	                        raise Exception("Could not get unique rule")
    54	                    node = GraphNode(
    55	                        unique_rule, GraphNodeMeta(stack_id, path_id, step_id), next_
    56	                    )
    57	                if node is None:
    58	                    raise Exception("Could not get node")

    77	    def _get_initial_pointers(self) -> "GenericSet[GraphPointer, str]":
    78	        pointers = make_pointers()
    79	
    80	        root_node = self._root_node
    81	        if not root_node:
    82	            raise Exception("Root node is not defined")
    83	
    84	        for node, parent in self._fetch_nodes_for_root_node(root_node):
    85	            pointer = GraphPointer(node, parent)
    86	            for resolved_pointer in self._resolve_pointer(pointer):
    87	                pointers.add(resolved_pointer)
    88	        return pointers

    90	    @staticmethod

    92	        for pointer in pointers:

    95	    def _parse(self, current_pointers, code_point: int):
    96	        for rule, graph_pointers in self._iterate_over_pointers(current_pointers):
    97	            if is_rule_char(rule):
    98	                valid = False
    99	                for possible_code_point in rule.value:
   100	                    if valid:
   101	                        break
   102	                    if is_range(possible_code_point):
   103	                        valid = is_point_in_range(code_point, possible_code_point)
   104	                    else:
   105	                        valid = code_point == possible_code_point
   106	                self._set_valid(graph_pointers, valid)
   107	            elif is_rule_char_excluded(rule):
   108	                valid = True
   109	                for possible_code_point in rule.value:
   110	                    if not valid:
   111	                        break
   112	                    if is_range(possible_code_point):
   113	                        valid = not is_point_in_range(code_point, possible_code_point)
   114	                    else:
   115	                        valid = code_point != possible_code_point
   116	                self._set_valid(graph_pointers, valid)
   117	            elif not is_rule_end(rule):
   118	                raise Exception(f"Unsupported rule: {rule!r}")
   119	
   120	        # a pointer's id is the sum of its node's id and its parent's id chain.
   121	        # if two pointers share the same id, it means they point to the same node and
   122	        # have identical parent chains. for the purposes of walking the graph, we only
   123	        # need to keep one of them.
   124	        next_pointers = make_pointers()
   125	        for current_pointer in current_pointers:
   126	            for unresolved_next_pointer in current_pointer.fetch_next():
   127	                for resolved_next_pointer in self._resolve_pointer(
   128	                    unresolved_next_pointer
   129	                ):
   130	                    next_pointers.add(resolved_next_pointer)
   131	        return next_pointers
   132	
   133	    def _resolve_pointer(
   134	        self, unresolved_pointer: GraphPointer
   135	    ) -> Iterator[GraphPointer]:
   136	        for resolved_pointer in unresolved_pointer.resolve():
   137	            if is_rule_ref(resolved_pointer.node.rule):
   138	                raise Exception(
   139	                    "Encountered a reference rule when building pointers to the graph"
   140	                )
   141	            if is_rule_end(resolved_pointer.node.rule) and resolved_pointer.parent:
   142	                raise Exception(
   143	                    "Encountered an ending rule with a parent when building pointers "
   144	                    "to the graph"
   145	                )
   146	            yield resolved_pointer
   147	
   148	    def add(self, src: ValidInput, _pointers=None):
   149	        pointers = _pointers if _pointers is not None else self._get_initial_pointers()
   150	        code_points = get_input_as_code_points(src)
   151	        for code_point_pos, code_point in enumerate(code_points):
   152	            pointers = self._parse(pointers, code_point)
   153	            if pointers.size == 0:
   154	                raise InputParseError(
   155	                    code_points, code_point_pos, list(self._previous_code_points)
   156	                )
   157	        self._previous_code_points.extend(code_points)
   158	        return pointers
   159	
   160	    def _fetch_nodes_for_root_node(
   161	        self,
   162	        root_nodes: RootNode,
   163	        parent: Optional[GraphPointer] = None,
   164	    ) -> Iterator[Tuple[GraphNode, Optional[GraphPointer]]]:
   165	        """Yield either the node, or if a reference rule, the referenced node.
   166	
   167	        We need this function, as distinct from leveraging the logic in GraphPointer,
   168	        because that needs a rule ref with already defined nodes; this function is used
   169	        to _set_ those nodes.
   170	        """
   171	        for node in root_nodes.values():
   172	            if is_rule_ref(node.rule):
   173	                yield from self._fetch_nodes_for_root_node(
   174	                    self._get_root_node(node.rule.value), GraphPointer(node, parent)
   175	                )
   176	            else:
   177	                yield node, parent

   179	    def print(self, pointers=None, colors: bool = False) -> str:
   180	        col = colorize if colors else no_colorize
   181	        graph_view: List[str] = []
   182	        for nodes in self.roots.values():
   183	            for node in nodes.values():
   184	                graph_view.append(
   185	                    node.print(pointers=pointers, show_position=True, colorize=col)
   186	                )
   187	        joined = "\n".join(graph_view)
   188	        return f"\n{joined}"

   190	    def __repr__(self) -> str:
   191	        return self.print(colors=True)

   193	    @staticmethod
   194	    def _iterate_over_pointers(pointers) -> Iterator[Tuple[object, List[GraphPointer]]]:
   195	        # keyed by rule identity: identical rules share a single reference, courtesy of
   196	        # the unique rules set built in the constructor.
   197	        seen_rules: Dict[int, Tuple[object, List[GraphPointer]]] = {}
   198	        for pointer in pointers:
   199	            rule = pointer.rule
   200	            if is_rule_ref(rule):
   201	                raise Exception(
   202	                    "Encountered a reference rule in the graph, this should not happen"
   203	                )
   204	            seen_rule = seen_rules.get(id(rule))
   205	            if seen_rule is None:
   206	                seen_rules[id(rule)] = (rule, [pointer])
   207	            else:
   208	                seen_rule[1].append(pointer)
   209	
   210	        for rule, graph_pointers in seen_rules.values():
   211	            yield rule, graph_pointers

    15	class GraphPointer:
    16	    def __init__(self, node: GraphNode, parent: Optional["GraphPointer"] = None):
    17	        if node is None:
    18	            raise Exception("Node is undefined")
    19	        self.node = node
    20	        self.parent = parent
    21	        self.id = f"{parent.id}-{node.id}" if parent is not None else node.id
    22	        self._valid: Optional[bool] = None
    23	
    24	    def resolve(self, resolved: bool = False) -> Iterator["GraphPointer"]:
    25	        """
    26	        1. If the current node is an end node, and the pointer has a parent, return the
    27	           parent's `fetch_next`; else return nothing.
    28	        2. If the current node is a rule ref, yield the referenced nodes, _unless_
    29	           resolved is true, in which case it returns next.
    30	        3. If the current node is a char or range, we go to the next node. If none
    31	           exists, throw an error.
    32	        """
    33	        if is_graph_pointer_rule_ref(self):
    34	            if resolved:
    35	                if self.node.next is None:
    36	                    raise Exception(f"No next node: {self.node!r}")
    37	                yield from GraphPointer(self.node.next, self.parent).resolve()
    38	            else:
    39	                for node in self.node.rule.nodes:
    40	                    yield from GraphPointer(node, self).resolve()
    41	        elif is_graph_pointer_rule_end(self):
    42	            if self.parent is None:
    43	                yield self
    44	            else:
    45	                yield from self.parent.resolve(True)
    46	        elif is_graph_pointer_rule_char(self) or is_graph_pointer_rule_char_exclude(
    47	            self
    48	        ):
    49	            yield self
    50	        else:
    51	            raise Exception(f"Unknown rule: {self.node.rule!r}")
    52	
    53	    def fetch_next(self) -> Iterator["GraphPointer"]:
    54	        # if this pointer is invalid, then we don't return any new pointers
    55	        if self._valid is False:
    56	            return
    57	
    58	        # if this pointer is an end node, we return the parent's next node. If no parent
    59	        # exists, we return nothing, since it's the end of the line.
    60	        if is_rule_end(self.node.rule):
    61	            if self.parent is not None:
    62	                yield from self.parent.fetch_next()
    63	        else:
    64	            if self.node.next is None:
    65	                raise Exception(f"No next node: {self.node!r}")
    66	            pointer = GraphPointer(self.node.next, self.parent)
    67	            yield from pointer.resolve()

    70	    def rule(self):
    71	        return self.node.rule

    73	    @property
    74	    def valid(self) -> Optional[bool]:
    75	        return self._valid

    78	    def valid(self, valid: bool) -> None:

    24	def build_rule_stack(linear_rules) -> Optional[List[List[UnresolvedRule]]]:
    25	    if linear_rules is None:
    26	        # the reference implementation's rules array can be sparse; holes map through
    27	        # untouched.
    28	        return None
    29	
    30	    paths: List[UnresolvedRule] = []
    31	    stack: List[List[UnresolvedRule]] = []
    32	
    33	    idx = 0
    34	    while idx < len(linear_rules):
    35	        rule_def = linear_rules[idx]
    36	        if is_rule_def_char(rule_def) or is_rule_def_char_not(rule_def):
    37	            # this could be a single char, or a range, or a sequence of alts; we don't
    38	            # know until we step through it.
    39	            char_rule = _make_char_rule(rule_def)
    40	            idx += 1
    41	            rule = linear_rules[idx] if idx < len(linear_rules) else None
    42	            while idx < len(linear_rules) and (
    43	                is_rule_def_char_rng_upper(rule) or is_rule_def_char_alt(rule)
    44	            ):
    45	                if is_rule_def_char_rng_upper(rule):
    46	                    # previous rule value should be a number
    47	                    if not char_rule.value:
    48	                        raise Exception("Unexpected undefined value")
    49	                    prev_value = char_rule.value.pop()
    50	                    if is_range(prev_value):
    51	                        raise Exception(
    52	                            "Unexpected range, expected a number but got an array: "
    53	                            f"{prev_value}"
    54	                        )
    55	                    char_rule.value.append([prev_value, rule.value])
    56	                if is_rule_def_char_alt(rule):
    57	                    char_rule.value.append(rule.value)
    58	                idx += 1
    59	                rule = linear_rules[idx] if idx < len(linear_rules) else None
    60	            paths.append(char_rule)
    61	        else:
    62	            if is_rule_def_alt(rule_def):
    63	                if not paths:
    64	                    raise Exception("Encountered alt without anything before it")
    65	                paths.append(RuleEnd())
    66	                stack.append(paths)
    67	                paths = []
    68	            elif is_rule_def_end(rule_def):
    69	                paths.append(RuleEnd())
    70	            elif is_rule_def_ref(rule_def):
    71	                paths.append(RuleRef(rule_def.value))
    72	            elif is_rule_def_char_alt(rule_def):
    73	                raise Exception(
    74	                    "Encountered char alt, should be handled by above block: "
    75	                    f"{rule_def.type}"
    76	                )
    77	            else:
    78	                raise Exception(f"Unsupported rule type: {rule_def.type}")
    79	            idx += 1
    80	
    81	    if not paths or not is_rule_end(paths[-1]):
    82	        paths.append(RuleEnd())
    83	
    84	    stack.append(paths)
    85	
    86	    return stack

    16	class RulesBuilder:
    17	    def __init__(self, src: str, limit: int = 1000):
    18	        self.pos = 0
    19	        self.symbol_ids = SymbolIds()
    20	        # rules are assigned by id, and ids are not necessarily assigned in order, so
    21	        # they're collected in a map and flattened out into a (possibly sparse) list.
    22	        self._rules_by_id: Dict[int, List[InternalRuleDef]] = {}
    23	        self.src = src

    25	        self._time_limit = limit
    26	        self.parse(src)
    27	
    28	    @property
    29	    def rules(self) -> List[Optional[List[InternalRuleDef]]]:
    30	        if not self._rules_by_id:
    31	            return []
    32	        length = max(self._rules_by_id) + 1
    33	        return [self._rules_by_id.get(idx) for idx in range(length)]
    34	
    35	    def _at(self, pos: int):
    36	        return self.src[pos] if 0 <= pos < len(self.src) else None
    37	
    38	    def parse(self, src: str) -> None:
    39	        # move cursor forward until we reach non-whitespace content
    40	        self.pos = parse_space(src, 0, True)
    41	
    42	        while self.pos < len(src):
    43	            self.parse_rule(src)
    44	
    45	        # Validate the state to ensure that all rules are defined
    46	        for rule in self.rules:
    47	            if rule is None:
    48	                continue
    49	            for elem in rule:
    50	                if elem.type == InternalRuleType.RULE_REF:
    51	                    # Ensure that the rule at that location exists
    52	                    referenced = self._rules_by_id.get(elem.value)
    53	                    rule_exists = referenced is not None and len(referenced) > 0
    54	                    if not rule_exists:
    55	                        missing_rule_name = self.symbol_ids.reverse_get(elem.value)
    56	                        missing_rule_pos = self.symbol_ids.get_pos(missing_rule_name)
    57	
    58	                        # Skip over the ::= and any whitespace
    59	                        while missing_rule_pos < len(src) and (
    60	                            src[missing_rule_pos] in (":", "=")
    61	                            or _WHITESPACE.search(src[missing_rule_pos])
    62	                        ):
    63	                            missing_rule_pos += 1
    64	
    65	                        raise GrammarParseError(
    66	                            src,
    67	                            missing_rule_pos,
    68	                            f'Undefined rule identifier "{missing_rule_name}"',
    69	                        )

   157	                    self.pos += inc_pos
   158	                    if type_ in (InternalRuleType.CHAR, InternalRuleType.CHAR_NOT):
   159	                        out_elements.append(
   160	                            InternalRuleDef(type_, [startchar_value])
   161	                        )
   162	                    else:
   163	                        out_elements.append(
   164	                            InternalRuleDef(type_, startchar_value)
   165	                        )
   166	
   167	                    if self._at(self.pos) == "-" and self._at(self.pos + 1) != "]":
   168	                        self.pos += 1
   169	                        endchar_value, inc_pos = parse_char(src, self.pos)
   170	                        out_elements.append(
   171	                            InternalRuleDef(
   172	                                InternalRuleType.CHAR_RNG_UPPER, endchar_value
   173	                            )
   174	                        )
   175	                        self.pos += inc_pos
   176	                self.pos = parse_space(src, self.pos + 1, is_nested)
   177	            elif is_word_char(char):

   179	                ref_rule_id = self.get_symbol_id(name, len(name))
   180	                self.pos += len(name)
   181	                self.pos = parse_space(src, self.pos, is_nested)
   182

   184	                out_elements.append(
   185	                    InternalRuleDef(InternalRuleType.RULE_REF, ref_rule_id)
   186	                )
   187	            elif char == "(":
   188	                self.pos = parse_space(src, self.pos + 1, True)
   189	                sub_rule_id = self.generate_symbol_id(rule_name)
   190	                self.parse_alternates(rule_name, sub_rule_id, depth + 1)
   191	                last_sym_start = len(out_elements)
   192	                out_elements.append(
   193	                    InternalRuleDef(InternalRuleType.RULE_REF, sub_rule_id)
   194	                )
   195	                if self._at(self.pos) != ")":
   196	                    raise GrammarParseError(
   197	                        src, self.pos, f"Expecting ')' at {self.pos}"
   198	                    )
   199	                self.pos = parse_space(src, self.pos + 1, is_nested)
   200	            elif char in ("*", "+", "?"):
   201	                if last_sym_start == len(out_elements):
   202	                    raise GrammarParseError(
   203	                        src,
   204	                        self.pos,
   205	                        f"Expecting preceding item to */+/? at {self.pos}",
   206	                    )
   207	                sub_rule_id = self.generate_symbol_id(rule_name)
   208	                sub_rule: List[InternalRuleDef] = out_elements[last_sym_start:]
   209	                if char in ("*", "+"):
   210	                    sub_rule.append(
   211	                        InternalRuleDef(InternalRuleType.RULE_REF, sub_rule_id)
   212	                    )
   213	                sub_rule.append(InternalRuleDef(InternalRuleType.ALT))
   214	                if char == "+":
   215	                    sub_rule.extend(out_elements[last_sym_start:])
   216	                sub_rule.append(InternalRuleDef(InternalRuleType.END))
   217	                self.add_rule(sub_rule_id, sub_rule)
   218	                del out_elements[last_sym_start:]
   219	                out_elements.append(
   220	                    InternalRuleDef(InternalRuleType.RULE_REF, sub_rule_id)
   221	                )
   222	                self.pos = parse_space(src, self.pos + 1, is_nested)
   223	            else:
   224	                break
   225	
   226	    def parse_alternates(
   227	        self,
   228	        rule_name: str,
   229	        rule_id: int,
   230	        depth: int = 0,
   231	    ) -> None:
   232	        src = self.src
   233	        rule: List[InternalRuleDef] = []
   234	        self.parse_sequence(rule_name, rule, depth)
   235	        while self._at(self.pos) == "|":
   236	            self.check_duration()
   237	            rule.append(InternalRuleDef(InternalRuleType.ALT))
   238	            self.pos = parse_space(src, self.pos + 1, True)
   239	            self.parse_sequence(rule_name, rule, depth)
   240	        rule.append(InternalRuleDef(InternalRuleType.END))
   241	        self.add_rule(rule_id, rule)

     1	from typing import List

     3	MAXIMUM_NUMBER_OF_ERROR_LINES_TO_SHOW = 3
     4	
     5	
     6	def build_error_position(src: str, pos: int) -> List[str]:
     7	    if src == "":
     8	        return ["No input provided"]
     9	
    10	    lines = src.split("\n")
    11	
    12	    line_idx = 0
    13	    while line_idx < len(lines) and lines[line_idx] and pos > len(lines[line_idx]) - 1:
    14	        pos -= len(lines[line_idx])
    15	        line_idx += 1
    16	
    17	    lines_to_show: List[str] = []
    18	    start = max(0, line_idx - (MAXIMUM_NUMBER_OF_ERROR_LINES_TO_SHOW - 1))
    19	    for i in range(start, line_idx + 1):
    20	        # the reference implementation indexes past the end of the array here, which
    21	        # joins as an empty string.
    22	        lines_to_show.append(lines[i] if i < len(lines) else "")
    23	
    24	    return [*lines_to_show, " " * max(0, pos) + "^"]
