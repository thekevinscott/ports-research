    94	    def _parse(self, current_pointers: GenericSet, code_point: int) -> GenericSet:
    95	        for rule, graph_pointers in self._iterate_over_pointers(current_pointers):
    96	            if is_rule_char(rule):
    97	                valid = False
    98	                for possible_code_point in rule.value:
    99	                    if is_range(possible_code_point):
   100	                        valid = is_point_in_range(code_point, possible_code_point)
   101	                    else:
   102	                        valid = code_point == possible_code_point
   103	                    if valid:
   104	                        break
   105	                self._set_valid(graph_pointers, valid)
   106	            elif is_rule_char_excluded(rule):
   107	                valid = True
   108	                for possible_code_point in rule.value:
   109	                    if is_range(possible_code_point):
   110	                        valid = not is_point_in_range(code_point, possible_code_point)
   111	                    else:
   112	                        valid = code_point != possible_code_point
   113	                    if not valid:
   114	                        break
   115	                self._set_valid(graph_pointers, valid)
   116	            elif not is_rule_end(rule):
   117	                raise Exception(f"Unsupported rule: {rule!r}")

   119	        # a pointer's id is the sum of its node's id and its parent's id chain.
   120	        # if two pointers share the same id, it means they point to the same node and have
   121	        # identical parent chains. for the purposes of walking the graph, we only need to
   122	        # keep one of them.
   123	        next_pointers = make_pointers()
   124	        for current_pointer in current_pointers:
   125	            for unresolved_next_pointer in current_pointer.fetch_next():
   126	                for resolved_next_pointer in self._resolve_pointer(unresolved_next_pointer):
   127	                    next_pointers.add(resolved_next_pointer)
   128	        return next_pointers

   130	    def _resolve_pointer(self, unresolved_pointer: GraphPointer) -> Iterator[GraphPointer]:

   132	            if is_rule_ref(resolved_pointer.node.rule):
   133	                raise Exception("Encountered a reference rule when building pointers to the graph")
   134	            if is_rule_end(resolved_pointer.node.rule) and resolved_pointer.parent is not None:

   141	        pointers = _pointers if _pointers is not None else self._get_initial_pointers()
   142	        code_points = get_input_as_code_points(src)
   143	        for code_point_pos, code_point in enumerate(code_points):

    15	class GraphPointer:
    16	    def __init__(self, node: GraphNode, parent: Optional["GraphPointer"] = None):
    17	        if node is None:
    18	            raise Exception("Node is undefined")
    19	        self.node = node
    20	        self.parent = parent
    21	        self.id = f"{parent.id}-{node.id}" if parent is not None else node.id
    22	        self._valid: Optional[bool] = None
    23	
    24	    def resolve(self, resolved: bool = False) -> Iterator["GraphPointer"]:
    25	        """
    26	        1. If the current node is an end node, and the pointer has a parent, return the
    27	           parent's `fetch_next`; else return nothing.
    28	        2. If the current node is a rule ref, yield the referenced nodes, _unless_ resolved
    29	           is true, in which case it returns next.
    30	        3. If the current node is a char or range, we go to the next node. If none exists,
    31	           throw an error.
    32	        """
    33	        if is_graph_pointer_rule_ref(self):
    34	            if resolved:
    35	                if self.node.next is None:
    36	                    raise Exception(f"No next node: {self.node!r}")
    37	                yield from GraphPointer(self.node.next, self.parent).resolve()
    38	            else:
    39	                for node in self.node.rule.nodes:
    40	                    yield from GraphPointer(node, self).resolve()
    41	        elif is_graph_pointer_rule_end(self):
    42	            if self.parent is None:
    43	                yield self
    44	            else:
    45	                yield from self.parent.resolve(True)
    46	        elif is_graph_pointer_rule_char(self) or is_graph_pointer_rule_char_exclude(self):
    47	            yield self
    48	        else:
    49	            raise Exception(f"Unknown rule: {self.node.rule!r}")
    50	
    51	    def fetch_next(self) -> Iterator["GraphPointer"]:
    52	        # if this pointer is invalid, then we don't return any new pointers
    53	        if self._valid is False:
    54	            return
    55	
    56	        # if this pointer is an end node, we return the parent's next node. If no parent
    57	        # exists, we return nothing, since it's the end of the line.
    58	        if is_rule_end(self.node.rule):
    59	            if self.parent is not None:
    60	                yield from self.parent.fetch_next()
    61	        else:
    62	            if self.node.next is None:
    63	                raise Exception(f"No next node: {self.node!r}")
    64	            pointer = GraphPointer(self.node.next, self.parent)
    65	            yield from pointer.resolve()
