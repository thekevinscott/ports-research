    11	class GenericSet(Generic[T, K]):
    12	    """A set whose membership is decided by a derived key, preserving insertion order."""
    13	
    14	    def __init__(self, get_key: Callable[[T], K]):
    15	        self.get_key = get_key
    16	        self._keys: dict[K, T] = {}
    17	        self._set: dict[int, T] = {}
    18	
    19	    def add(self, el: T) -> None:
    20	        key = self.get_key(el)
    21	        if key not in self._keys:
    22	            self._keys[key] = el
    23	            self._set[id(el)] = el

    25	    def delete(self, el: T) -> None:
    26	        key = self.get_key(el)
    27	        ref = self._keys.get(key)

    31	class Graph:
    32	    def __init__(self, grammar: str, stacked_rules: list, root_id: int):
    33	        self.roots: dict[int, RootNode] = {}
    34	        self.grammar = grammar
    35	        self._previous_code_points: list[int] = []
    36	
    37	        rule_refs: list[RuleRef] = []
    38	        unique_rules = GenericSet(get_serialized_rule_key)
    39	        for stack_id in range(len(stacked_rules)):
    40	            stack = stacked_rules[stack_id]
    41	            nodes: dict[int, GraphNode] = {}
    42	            for path_id in range(len(stack)):
    43	                path = stack[path_id]
    44	                node: Optional[GraphNode] = None
    45	                for step_id in range(len(path) - 1, -1, -1):
    46	                    next_node = node
    47	                    rule = stack[path_id][step_id]
    48	                    unique_rules.add(rule)
    49	                    if is_rule_ref(rule):
    50	                        rule_refs.append(rule)
    51	                    # rules coming in may be identical but have different references.
    52	                    # here, we ensure we always use the same reference for an identical rule.
    53	                    # this makes future comparisons easier.
    54	                    unique_rule = unique_rules.get(rule)
    55	                    if unique_rule is None:
    56	                        raise GBNFError("Could not get unique rule")
    57	                    node = GraphNode(
    58	                        unique_rule, GraphNodeMeta(stack_id, path_id, step_id), next_node
    59	                    )
    60	                if node is None:
    61	                    raise GBNFError("Could not get node")
    62	                nodes[path_id] = node
    63	
    64	            self.roots[stack_id] = nodes
    65	
    66	        root_node = self.roots.get(root_id)
    67	        if not root_node:
    68	            raise GBNFError(f"Root node not found for value: {root_id}")
    69	        self._root_node = root_node
    70	
    71	        for rule_ref in rule_refs:
    72	            referenced_nodes: list[GraphNode] = []
    73	            for node in self._get_root_node(rule_ref.value).values():
    74	                if node not in referenced_nodes:
    75	                    referenced_nodes.append(node)
    76	            rule_ref.nodes = referenced_nodes

   103	        for pointer in pointers:

   106	    def _parse(self, current_pointers: GenericSet, code_point: int) -> GenericSet:
   107	        for rule, graph_pointers in self._iterate_over_pointers(current_pointers):
   108	            if is_rule_char(rule):
   109	                valid = False
   110	                for possible_code_point in rule.value:
   111	                    if valid:
   112	                        break
   113	                    if is_range(possible_code_point):
   114	                        valid = is_point_in_range(code_point, possible_code_point)
   115	                    else:
   116	                        valid = code_point == possible_code_point
   117	                self._set_valid(graph_pointers, valid)
   118	            elif is_rule_char_excluded(rule):
   119	                valid = True
   120	                for possible_code_point in rule.value:
   121	                    if not valid:
   122	                        break
   123	                    if is_range(possible_code_point):
   124	                        valid = not is_point_in_range(code_point, possible_code_point)
   125	                    else:
   126	                        valid = code_point != possible_code_point
   127	                self._set_valid(graph_pointers, valid)
   128	            elif not is_rule_end(rule):
   129	                raise GBNFError(f"Unsupported rule: {rule}")
   130	
   131	        # a pointer's id is the sum of its node's id and its parent's id chain.
   132	        # if two pointers share the same id, it means they point to the same node and have
   133	        # identical parent chains. for the purposes of walking the graph, we only need to
   134	        # keep one of them.
   135	        next_pointers = make_pointers()
   136	        for current_pointer in current_pointers:
   137	            for unresolved_next_pointer in current_pointer.fetch_next():
   138	                for resolved_next_pointer in self._resolve_pointer(unresolved_next_pointer):
   139	                    next_pointers.add(resolved_next_pointer)
   140	        return next_pointers
   141	
   142	    def _resolve_pointer(self, unresolved_pointer: GraphPointer) -> Iterator[GraphPointer]:
   143	        for resolved_pointer in unresolved_pointer.resolve():
   144	            if is_rule_ref(resolved_pointer.node.rule):
   145	                raise GBNFError(
   146	                    "Encountered a reference rule when building pointers to the graph"
   147	                )
   148	            if is_rule_end(resolved_pointer.node.rule) and resolved_pointer.parent:
   149	                raise GBNFError(
   150	                    "Encountered an ending rule with a parent when building pointers to "
   151	                    "the graph"
   152	                )
   153	            yield resolved_pointer
   154	
   155	    def add(self, src, pointers: Optional[GenericSet] = None) -> GenericSet:
   156	        current = pointers if pointers is not None else self._get_initial_pointers()
   157	        code_points = get_input_as_code_points(src)
   158	        for code_point_pos in range(len(code_points)):
   159	            code_point = code_points[code_point_pos]
   160	            current = self._parse(current, code_point)
   161	            if current.size == 0:
   162	                raise InputParseError(code_points, code_point_pos, self._previous_code_points)
   163	        self._previous_code_points.extend(code_points)
   164	        return current
   165	
   166	    # generator that yields either the node, or if a reference rule, the referenced node
   167	    # we need this function, as distinct from leveraging the logic in GraphPointer,
   168	    # because that needs a rule ref with already defined nodes; this function is used to
   169	    # _set_ those nodes
   170	    def _fetch_nodes_for_root_node(
   171	        self,
   172	        root_nodes: dict,
   173	        parent: Optional[GraphPointer] = None,
   174	    ) -> Iterator[tuple[GraphNode, Optional[GraphPointer]]]:
   175	        for node in list(root_nodes.values()):
   176	            if is_rule_ref(node.rule):
   177	                yield from self._fetch_nodes_for_root_node(
   178	                    self._get_root_node(node.rule.value), GraphPointer(node, parent)
   179	                )
   180	            else:
   181	                yield node, parent
   182	
   183	    # --------------------------------------------------------------- printing
   184	
   185	    def __repr__(self) -> str:
   186	        return self.print(colors=True)
   187	
   188	    def print(self, pointers: Optional[GenericSet] = None, colors: bool = False) -> str:
   189	        nodes = [list(root.values()) for root in self.roots.values()]
   190	        graph_view: list[str] = []
   191	        for root_node in nodes:
   192	            for node in root_node:
   193	                graph_view.append(
   194	                    node.print(
   195	                        pointers=pointers,
   196	                        show_position=True,
   197	                        colorize=colorize if colors else no_color,
   198	                    )
   199	                )
   200	        return "\n" + "\n".join(graph_view)

   202	    def _iterate_over_pointers(
   203	        self, pointers: GenericSet
   204	    ) -> Iterator[tuple[object, list[GraphPointer]]]:
   205	        seen_rules: dict[int, tuple[object, list[GraphPointer]]] = {}
   206	        for pointer in pointers:
   207	            rule = pointer.rule
   208	            if is_rule_ref(rule):
   209	                raise GBNFError(
   210	                    "Encountered a reference rule in the graph, this should not happen"
   211	                )
   212	            seen_rule = seen_rules.get(id(rule))
   213	            if seen_rule is None:
   214	                seen_rules[id(rule)] = (rule, [pointer])
   215	            else:
   216	                seen_rule[1].append(pointer)
   217	
   218	        yield from seen_rules.values()

    18	class GraphPointer:
    19	    def __init__(self, node: GraphNode, parent: Optional["GraphPointer"] = None):
    20	        if node is None:
    21	            raise GBNFError("Node is undefined")
    22	        self.node = node
    23	        self.parent = parent
    24	        self._valid: Optional[bool] = None
    25	        self.id = f"{parent.id}-{node.id}" if parent else node.id
    26	
    27	    def resolve(self, resolved: bool = False) -> Iterator["GraphPointer"]:
    28	        """
    29	        1. If the current node is an end node, and the pointer has a parent, return the
    30	           parent's `fetch_next`; else return nothing.
    31	        2. If the current node is a rule ref, yield the referenced nodes, _unless_ resolved
    32	           is true, in which case it returns next.
    33	        3. If the current node is a char or range, we go to the next node. If none exists,
    34	           throw an error.
    35	        """
    36	        if is_graph_pointer_rule_ref(self):
    37	            if resolved:
    38	                if self.node.next is None:
    39	                    raise GBNFError(f"No next node: {self.node}")
    40	                yield from GraphPointer(self.node.next, self.parent).resolve()
    41	            else:
    42	                for node in self.node.rule.nodes:
    43	                    yield from GraphPointer(node, self).resolve()
    44	        elif is_graph_pointer_rule_end(self):
    45	            if not self.parent:
    46	                yield self
    47	            else:
    48	                yield from self.parent.resolve(True)
    49	        elif is_graph_pointer_rule_char(self) or is_graph_pointer_rule_char_exclude(self):
    50	            yield self
    51	        else:
    52	            raise GBNFError(f"Unknown rule: {self.node.rule}")
    53	
    54	    def fetch_next(self) -> Iterator["GraphPointer"]:
    55	        # if this pointer is invalid, then we don't return any new pointers
    56	        if self._valid is False:
    57	            return
    58	
    59	        # if this pointer is an end node, we return the parent's next node. If no parent
    60	        # exists, we return nothing, since it's the end of the line.
    61	        if is_rule_end(self.node.rule):
    62	            if self.parent:
    63	                yield from self.parent.fetch_next()
    64	        else:
    65	            if self.node.next is None:
    66	                raise GBNFError(f"No next node: {self.node}")
    67	            pointer = GraphPointer(self.node.next, self.parent)
    68	            yield from pointer.resolve()

    70	    @property
    71	    def rule(self):
    72	        return self.node.rule
    73	
    74	    @property
    75	    def valid(self) -> Optional[bool]:
    76	        return self._valid

    79	    def valid(self, valid: bool) -> None:

    26	        seen: set[str] = set()
    27	        for pointer in self._pointers:
    28	            rule = pointer.rule
    29	            key = rule.to_json()
    30	            if key not in seen:
    31	                seen.add(key)
    32	                yield rule

    35	        pointers = self._graph.add(input, self._pointers)

    13	def print_graph_node(node, colorize, pointers=None, show_position: bool = False) -> str:
    14	    col = colorize
    15	    rule = node.rule
    16	
    17	    parts: list[str] = []
    18	    if show_position:
    19	        parts.extend(
    20	            [
    21	                col("{", Color.BLUE),
    22	                col(node.id, Color.GRAY),
    23	                col("}", Color.BLUE),
    24	            ]
    25	        )
    26	    if is_rule_char(rule):
    27	        parts.extend(
    28	            [
    29	                col("[", Color.GRAY),
    30	                col(
    31	                    "".join(
    32	                        # a range renders as an array in JS, which `join` stringifies
    33	                        # with commas
    34	                        ",".join(col(from_char_code(val), Color.YELLOW) for val in v)
    35	                        if is_range(v)
    36	                        else _get_char(v)
    37	                        for v in rule.value
    38	                    ),
    39	                    Color.YELLOW,
    40	                ),
    41	                col("]", Color.GRAY),
    42	            ]
    43	        )
    44	    elif is_rule_ref(rule):
    45	        parts.append(
    46	            col("Ref(", Color.GRAY) + col(f"{rule.value}", Color.GREEN) + col(")", Color.GRAY)
    47	        )
    48	    else:
    49	        parts.append(col(rule.type.value, Color.YELLOW))
    50	
    51	    if pointers:
    52	        for pointer in pointers:
    53	            pointer_parts: list[str] = []
    54	            if pointer.node is node:

    19	def _json_value(value) -> str:
    20	    """Serialize a rule value the way ``JSON.stringify`` would."""
    21	    if isinstance(value, (list, tuple)):
    22	        return "[" + ",".join(_json_value(item) for item in value) + "]"
