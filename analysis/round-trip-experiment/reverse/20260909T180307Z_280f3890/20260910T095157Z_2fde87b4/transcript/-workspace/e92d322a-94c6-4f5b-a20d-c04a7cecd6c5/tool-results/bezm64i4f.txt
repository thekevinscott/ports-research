    23	 * Parse a single (possibly escaped) character out of `src` at `pos`.
    24	 *

    28	  if (pos >= src.length) {
    29	    throw new GrammarParseError(
    30	      src,

    33	    );

    38	    if (next === 'x') {

    41	    if (next === 'u') {
    42	      return [parseHex(src, pos + 2, pos + 6, pos), 6];
    43	    }

    55	  // indexing is by code point.
    56	  return [codePoint, codePoint > 0xffff ? 2 : 1];

   161	  parseSequence(ruleName: string, outElements: InternalRuleDef[], depth = 0): void {
   162	    const isNested = depth !== 0;
   163	    const src = this.src;
   164	    let lastSymStart = outElements.length;
   165

   185	        while (src[this.pos] !== ']') {
   186	          this.checkDuration();

   189	          const [startcharValue, incPos] = parseChar(src, this.pos);
   190	          this.pos += incPos;
   191	          outElements.push(getOutElements(type, startcharValue));
   192	
   193	          if (src[this.pos] === '-' && src[this.pos + 1] !== ']') {
   194	            this.pos += 1;

   196	            outElements.push(new InternalRuleDefCharRngUpper(endcharValue));
   197	            this.pos += endIncPos;
   198	          }
   199	        }
   200	        this.pos = parseSpace(src, this.pos + 1, isNested);
   201	      } else if (isWordChar(src[this.pos] as string)) {
   202	        const name = parseName(src, this.pos);
   203	        const refRuleId = this.getSymbolId(name, name.length);
   204	        this.pos += name.length;
   205	        this.pos = parseSpace(src, this.pos, isNested);
   206	
   207	        lastSymStart = outElements.length;
   208	        outElements.push(new InternalRuleDefReference(refRuleId));
   209	      } else if (src[this.pos] === '(') {
   210	        this.pos = parseSpace(src, this.pos + 1, true);
   211	        const subRuleId = this.generateSymbolId(ruleName);
   212	        this.parseAlternates(ruleName, subRuleId, depth + 1);
   213	        lastSymStart = outElements.length;
   214	        outElements.push(new InternalRuleDefReference(subRuleId));
   215	        if (src[this.pos] !== ')') {
   216	          throw new GrammarParseError(src, this.pos, `Expecting ')' at ${this.pos}`);
   217	        }
   218	        this.pos = parseSpace(src, this.pos + 1, isNested);
   219	      } else if (src[this.pos] === '*' || src[this.pos] === '+' || src[this.pos] === '?') {
   220	        if (lastSymStart === outElements.length) {
   221	          throw new GrammarParseError(
   222	            src,
   223	            this.pos,
   224	            `Expecting preceding item to */+/? at ${this.pos}`,
   225	          );
   226	        }
   227	        const subRuleId = this.generateSymbolId(ruleName);
   228	        const subRule = outElements.slice(lastSymStart);
   229	        if (src[this.pos] === '*' || src[this.pos] === '+') {
   230	          subRule.push(new InternalRuleDefReference(subRuleId));
   231	        }
   232	        subRule.push(new InternalRuleDefAlt());
   233	        if (src[this.pos] === '+') {
   234	          subRule.push(...outElements.slice(lastSymStart));
   235	        }
   236	        subRule.push(new InternalRuleDefEnd());
   237	        this.addRule(subRuleId, subRule);
   238	        outElements.splice(
   239	          lastSymStart,
   240	          outElements.length - lastSymStart,
   241	          new InternalRuleDefReference(subRuleId),
   242	        );
   243	        this.pos = parseSpace(src, this.pos + 1, isNested);
   244	      } else {
   245	        break;
   246	      }
   247	    }
   248	  }

   250	  parseAlternates(ruleName: string, ruleId: number, depth = 0): void {
   251	    const src = this.src;
   252	    const rule: InternalRuleDef[] = [];
   253	    this.parseSequence(ruleName, rule, depth);
   254	    // Ensure that this.pos is within bounds before checking src[this.pos]
   255	    while (this.pos < src.length && src[this.pos] === '|') {
   256	      this.checkDuration();
   257	      rule.push(new InternalRuleDefAlt());
   258	      this.pos = parseSpace(src, this.pos + 1, true);
   259	      this.parseSequence(ruleName, rule, depth);
   260	    }
   261	    rule.push(new InternalRuleDefEnd());
   262	    this.addRule(ruleId, rule);
