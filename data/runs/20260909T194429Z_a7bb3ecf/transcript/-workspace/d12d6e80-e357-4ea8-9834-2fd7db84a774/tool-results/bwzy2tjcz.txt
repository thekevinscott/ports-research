     1	import { colorize, } from "./colorize.js";
     2	import { GraphNode, } from "./graph-node.js";
     3	import { printGraphPointer, } from "./print.js";
     4	import {
     5	  UnresolvedRule,
     6	  customInspectSymbol,
     7	  type ResolvedGraphPointer,
     8	} from "./types.js";
     9	import {
    10	  isRuleEnd,
    11	  isGraphPointerRuleRef,
    12	  isGraphPointerRuleEnd,
    13	  isGraphPointerRuleChar,
    14	  isGraphPointerRuleCharExclude,
    15	} from './type-guards.js';
    16	
    17	
    18	export class GraphPointer<R extends UnresolvedRule = UnresolvedRule> {
    19	  node: GraphNode<R>;
    20	  parent?: GraphPointer;
    21	  #valid?: boolean;
    22	  id: string;
    23	
    24	  constructor(node: GraphNode<R>, parent?: GraphPointer) {
    25	    if (node === undefined) {
    26	      throw new Error('Node is undefined');
    27	    }
    28	    this.node = node;
    29	    this.parent = parent;
    30	    this.id = parent ? `${parent.id}-${node.id}` : node.id;
    31	  }
    32	
    33	  *resolve(resolved = false): IterableIterator<ResolvedGraphPointer> {
    34	    /*
    35	     * 1. If the current node is an end node, and the pointer has a parent, return the parent's `fetchNext`; else return nothing.
    36	     * 2. If the current node is a rule ref, yield the referenced nodes, _unless_ resolved is true, in which case it returns next.
    37	     * 3. If the current node is a char or range, we go to the next node. If none exists, throw an error.
    38	     */
    39	    if (isGraphPointerRuleRef(this)) {
    40	      if (resolved) {
    41	        if (!this.node.next) {
    42	          throw new Error(`No next node: ${JSON.stringify(this.node)}`);
    43	        }
    44	        yield* new GraphPointer(this.node.next, this.parent).resolve();
    45	      } else {
    46	
    47	        for (const node of this.node.rule.nodes) {
    48	          yield* new GraphPointer(node, this).resolve();
    49	        }
    50	      }
    51	    } else if (isGraphPointerRuleEnd(this)) {
    52	      if (!this.parent) {
    53	        yield this;
    54	      } else {
    55	        yield* this.parent.resolve(true);
    56	      }
    57	    } else if (isGraphPointerRuleChar(this) || isGraphPointerRuleCharExclude(this)) {
    58	      yield this;
    59	    } else {
    60	      throw new Error(`Unknown rule: ${JSON.stringify(this.node.rule)}`);
    61	    }
    62	  }
    63	
    64	  *fetchNext(): IterableIterator<ResolvedGraphPointer> {
    65	    // if this pointer is invalid, then we don't return any new pointers
    66	    if (this.#valid === false) {
    67	      return;
    68	    }
    69	
    70	    // if this pointer is an end node, we return the parent's next node. If no parent exists,
    71	    // we return nothing, since it's the end of the line.
    72	    if (isRuleEnd(this.node.rule)) {
    73	      if (this.parent) {
    74	        yield* this.parent.fetchNext();
    75	      }
    76	    } else {
    77	      if (!this.node.next) {
    78	        throw new Error(`No next node: ${JSON.stringify(this.node)}`);
    79	      }
    80	      const pointer = new GraphPointer(this.node.next, this.parent);
    81	      yield* pointer.resolve();
    82	    }
    83	  }
    84	
    85	  get rule() {
    86	    return this.node.rule;
    87	  }
    88	
    89	  set valid(valid: boolean) {
    90	    this.#valid = valid;
    91	  }
    92	
    93	  print = printGraphPointer(this);
    94	
    95	  [customInspectSymbol]() {
    96	    return this.print({ colorize, });
    97	  }
    98	}

     2	import { GraphNode, } from "./graph-node.js";

     7	  type UnresolvedRule,

     9	  customInspectSymbol,
    10	  type ValidInput,
    11	  type ResolvedGraphPointer,
    12	} from "./types.js";
    13	import {
    14	  isRange,
    15	  isRuleChar,
    16	  isRuleCharExcluded,
    17	  isRuleEnd,
    18	  isRuleRef,
    19	} from './type-guards.js';
    20	import { isPointInRange, } from "../utils/is-point-in-range.js";
    21	import {
    22	  InputParseError,
    23	} from "../utils/errors/input-parse-error.js";
    24	import { RuleRef, } from "./rule-ref.js";
    25	import { getInputAsCodePoints, } from "./get-input-as-code-points.js";
    26	
    27	type RootNode = Map<number, GraphNode>;
    28	const makePointers = () => new GenericSet<ResolvedGraphPointer, string>(p => p.id);
    29	export class Graph {
    30	  private roots = new Map<number, RootNode>();
    31	  #rootNode?: RootNode;
    32	  grammar: string;
    33	  private previousCodePoints: number[] = [];
    34	
    35	  constructor(grammar: string, stackedRules: UnresolvedRule[][][], rootId: number) {
    36	    this.grammar = grammar;
    37	    const ruleRefs: RuleRef[] = [];
    38	    const uniqueRules = new GenericSet<UnresolvedRule, string>(rule => getSerializedRuleKey(rule));
    39	    for (let stackId = 0; stackId < stackedRules.length; stackId++) {
    40	      const stack = stackedRules[stackId];
    41	      const nodes = new Map<number, GraphNode>();
    42	      for (let pathId = 0; pathId < stack.length; pathId++) {
    43	        const path = stack[pathId];
    44	        let node: GraphNode | undefined = undefined;
    45	        for (let stepId = path.length - 1; stepId >= 0; stepId--) {
    46	          const next: undefined | GraphNode = node;
    47	          const rule = stack[pathId][stepId];
    48	          uniqueRules.add(rule);
    49	          if (isRuleRef(rule)) {
    50	            ruleRefs.push(rule);
    51	          }
    52	          // rules coming in may be identical but have different references.
    53	          // here, we ensure we always use the same reference for an identical rule.
    54	          // this makes future comparisons easier.
    55	          const uniqueRule = uniqueRules.get(rule);
    56	          if (uniqueRule === undefined) {
    57	            throw new Error('Could not get unique rule');
    58	          }
    59	          node = new GraphNode(uniqueRule, { stackId, pathId, stepId, }, next);
    60	        }
    61	        if (node === undefined) {
    62	          throw new Error('Could not get node');
    63	        }
    64	        nodes.set(pathId, node);
    65	      }
    66	
    67	      this.roots.set(stackId, nodes);
    68	    }
    69	    const __rootNode = this.roots.get(rootId);
    70	    if (!__rootNode) {
    71	      throw new Error(`Root node not found for value: ${rootId}`);
    72	    }
    73	    this.#rootNode = __rootNode;
    74	
    75	    for (const ruleRef of ruleRefs) {
    76	      const referencedNodes = new Set<GraphNode>();
    77	      for (const node of this.getRootNode(ruleRef.value).values()) {
    78	        referencedNodes.add(node);
    79	      }
    80	      ruleRef.nodes = referencedNodes;
    81	    }
    82	  }
    83	
    84	  private getRootNode = (value: number): RootNode => {
    85	    const rootNode = this.roots.get(value);
    86	    if (!rootNode) {
    87	      throw new Error(`Root node not found for value: ${value}`);
    88	    }
    89	    return rootNode;
    90	  };
    91	
    92	
    93	  private getInitialPointers = (): Pointers => {
    94	    const pointers = makePointers();
    95	
    96	    const rootNode = this.#rootNode;
    97	    if (!rootNode) {
    98	      throw new Error('Root node is not defined');
    99	    }
   100	
   101	    for (const { node, parent, } of this.fetchNodesForRootNode(rootNode)) {
   102	      const pointer = new GraphPointer(node, parent);
   103	      for (const resolvedPointer of this.resolvePointer(pointer)) {
   104	        pointers.add(resolvedPointer);
   105	      }
   106	    }
   107	    return pointers;
   108	  };
   109	
   110	  private setValid(pointers: GraphPointer[], valid: boolean) {
   111	    for (const pointer of pointers) {
   112	      pointer.valid = valid;
   113	    }
   114	  }
   115	
   116	  private parse(currentPointers: Pointers, codePoint: number): Pointers {
   117	    for (const { rule, graphPointers, } of this.iterateOverPointers(currentPointers)) {
   118	      if (isRuleChar(rule)) {
   119	        const valid = rule.value.reduce((
   120	          isValid,
   121	          possibleCodePoint,
   122	        ) => {
   123	          if (isValid) {
   124	            return true;
   125	          }
   126	          if (isRange(possibleCodePoint)) {
   127	            return isPointInRange(codePoint, possibleCodePoint);
   128	          }
   129	          return codePoint === possibleCodePoint;
   130	        }, false);
   131	        this.setValid(graphPointers, valid);
   132	      } else if (isRuleCharExcluded(rule)) {
   133	        const valid = rule.value.reduce((
   134	          isValid,
   135	          possibleCodePoint,
   136	        ) => {
   137	          if (!isValid) {
   138	            return false;
   139	          }
   140	          return isRange(possibleCodePoint) ? !isPointInRange(codePoint, possibleCodePoint) : codePoint !== possibleCodePoint;
   141	        }, true);
   142	        this.setValid(graphPointers, valid);
   143	      } else if (!isRuleEnd(rule)) {
   144	        throw new Error(`Unsupported rule: ${JSON.stringify(rule)}`);
   145	      }
   146	    }
   147	
   148	    // a pointer's id is the sum of its node's id and its parent's id chain.
   149	    // if two pointers share the same id, it means they point to the same node and have identical parent chains.
   150	    // for the purposes of walking the graph, we only need to keep one of them.
   151	    const nextPointers = makePointers();
   152	    for (const currentPointer of currentPointers) {
   153	      for (const unresolvedNextPointer of currentPointer.fetchNext()) {
   154	        for (const resolvedNextPointer of this.resolvePointer(unresolvedNextPointer)) {
   155	          nextPointers.add(resolvedNextPointer);
   156	        }
   157	      }
   158	    }
   159	    return nextPointers;
   160	  }
   161	
   162	  private * resolvePointer(unresolvedPointer: GraphPointer): IterableIterator<ResolvedGraphPointer> {
   163	    for (const resolvedPointer of unresolvedPointer.resolve()) {
   164	      if (isRuleRef(resolvedPointer.node.rule)) {
   165	        throw new Error('Encountered a reference rule when building pointers to the graph');
   166	      }
   167	      if (isRuleEnd(resolvedPointer.node.rule) && !!resolvedPointer.parent) {
   168	        throw new Error('Encountered an ending rule with a parent when building pointers to the graph');
   169	      }
   170	      yield resolvedPointer;
   171	    }
   172	  }
   173	
   174	  public add = (src: ValidInput, _pointers?: Pointers,): Pointers => {
   175	    let pointers = _pointers || this.getInitialPointers();
   176	    const codePoints = getInputAsCodePoints(src);
   177	    for (let codePointPos = 0; codePointPos < codePoints.length; codePointPos++) {
   178	      const codePoint = codePoints[codePointPos];
   179	      pointers = this.parse(pointers, codePoint);
   180	      if (pointers.size === 0) {
   181	        throw new InputParseError(codePoints, codePointPos, this.previousCodePoints);
   182	      }
   183	    }
   184	    this.previousCodePoints.push(...codePoints);
   185	    return pointers;
   186	  };
   187	
   188	  // generator that yields either the node, or if a reference rule, the referenced node
   189	  // we need these function, as distinct from leveraging the logic in GraphPointer,
   190	  // because that needs a rule ref with already defined nodes; this function is used to _set_ those nodes
   191	  private * fetchNodesForRootNode(
   192	    rootNodes: Map<number, GraphNode>,
   193	    parent?: GraphPointer,
   194	  ): IterableIterator<{ node: GraphNode; parent?: GraphPointer; }> {
   195	    for (const node of rootNodes.values()) {
   196	      if (isRuleRef(node.rule)) {
   197	        yield* this.fetchNodesForRootNode(this.getRootNode(node.rule.value), new GraphPointer(node, parent));
   198	      } else {
   199	        yield { node, parent, };
   200	      }
   201	    }
   202	  }
   203	
   204	
   205	  [customInspectSymbol](
   206	    // depth: number, inspectOptions: InspectOptions, inspect: CustomInspectFunction
   207	  ) {
   208	    return this.print({ colors: true, });
   209	  }
   210	
   211	  public print = ({ pointers, colors = false, }: { pointers?: Pointers; colors?: boolean } = {}) => {
   212	    const nodes: GraphNode[][] = Array.from(this.roots.values()).map(nodes => Array.from(nodes.values()));
   213	    const graphView = nodes.reduce<string[]>((acc, rootNode) => acc.concat(rootNode.map(node => node.print({
   214	      pointers,
   215	      showPosition: true,
   216	      colorize: colors ? colorize : str => `${str}`,
   217	    }))), []);
   218	    return `\n${graphView.join('\n')}`;
   219	  };
   220	
   221	  private * iterateOverPointers(pointers: Pointers): IterableIterator<{ rule: UnresolvedRule; graphPointers: GraphPointer[]; }> {
   222	    const seenRules = new Map<UnresolvedRule, GraphPointer[]>();
   223	    for (const pointer of pointers) {
   224	      const rule = pointer.rule;
   225	      if (isRuleRef(rule)) {
   226	        throw new Error('Encountered a reference rule in the graph, this should not happen');
   227	      }
   228	      let seenRule = seenRules.get(rule);
   229	      if (!seenRule) {
   230	        seenRule = [pointer,];
   231	        seenRules.set(rule, seenRule,);
   232	      } else {
   233	        seenRule.push(pointer);
   234	      }
   235	    }
   236	
   237	    for (const [rule, graphPointers,] of seenRules.entries()) {
   238	      yield { rule, graphPointers, };
   239	    }
   240	  }
   241	}

    28	function makeCharRule<R extends (InternalRuleDefChar | InternalRuleDefCharNot)>(ruleDef: R): R extends InternalRuleDefChar ? RuleChar : RuleCharExclude {
    29	  return {
    30	    type: isRuleDefCharNot(ruleDef) ? RuleType.CHAR_EXCLUDE : RuleType.CHAR,

    35	export const buildRuleStack = (linearRules: InternalRuleDef[]): UnresolvedRule[][] => {
    36	  let paths: UnresolvedRule[] = [];
    37	
    38	  const stack: UnresolvedRule[][] = [];
    39	
    40	  let idx = 0;
    41	  while (idx < linearRules.length) {
    42	    const ruleDef = linearRules[idx];
    43	    if (isRuleDefChar(ruleDef) || isRuleDefCharNot(ruleDef)) {
    44	      // this could be a single char, or a range, or a sequence of alts; we don't know until we step through it.
    45	      const charRule = makeCharRule(ruleDef);
    46	      idx += 1;
    47	      let rule = linearRules[idx];
    48	      while (idx < linearRules.length && (isRuleDefCharRngUpper(rule) || isRuleDefCharAlt(rule))) {
    49	        if (isRuleDefCharRngUpper(rule)) {
    50	          // previous rule value should be a number
    51	          const prevValue = charRule.value.pop();
    52	          if (isRange(prevValue)) {
    53	            throw new Error(`Unexpected range, expected a number but got an array: ${JSON.stringify(prevValue)}`);
    54	          }
    55	          if (prevValue === undefined) {
    56	            throw new Error('Unexpected undefined value');
    57	          }
    58	          charRule.value.push([
    59	            prevValue,
    60	            rule.value,
    61	          ]);
    62	        }
    63	        if (isRuleDefCharAlt(rule)) {
    64	          charRule.value.push(rule.value);
    65	        }
    66	        idx += 1;
    67	        rule = linearRules[idx];
    68	      }
    69	      paths.push(charRule);
    70	    } else {
    71	      if (isRuleDefAlt(ruleDef)) {
    72	        if (!paths.length) {
    73	          throw new Error('Encountered alt without anything before it');
    74	        }
    75	        paths.push({ type: RuleType.END, });
    76	        stack.push(paths);
    77	        paths = [];
    78	      } else if (isRuleDefEnd(ruleDef)) {
    79	        paths.push({
    80	          type: RuleType.END,
    81	        });
    82	      } else if (isRuleDefRef(ruleDef)) {
    83	        paths.push(new RuleRef(ruleDef.value));
    84	      } else if (isRuleDefCharAlt(ruleDef)) {
    85	        throw new Error(`Encountered char alt, should be handled by above block: ${ruleDef.type}`);
    86	      } else {
    87	        throw new Error(`Unsupported rule type: ${ruleDef.type}`);
    88	      }
    89	      idx += 1;
    90	    }
    91	
    92	  }
    93	  if (!isRuleEnd(paths[paths.length - 1])) {
    94	    paths.push({ type: RuleType.END, });
    95	  }
    96	
    97	  stack.push(paths);
    98	
    99	  return stack;
   100	};

     3	export const parseChar = (src: string, pos: number): [number, number] => {
     4	  if (src[pos] === '\\') {
     5	    // if (src.slice(pos, pos + 1) === '\\') {
     6	    switch (src[pos + 1]) {
     7	      case 'x': return [parseInt(src.slice(pos + 2, pos + 4), 16), 4,];
     8	      case 'u': return [parseInt(src.slice(pos + 2, pos + 6), 16), 6,];
     9	      case 'U': return [parseInt(src.slice(pos + 2, pos + 10), 16), 10,];
    10	      case 't': return ['\t'.charCodeAt(0), 2,];
    11	      case 'r': return ['\r'.charCodeAt(0), 2,];
    12	      case 'n': return ['\n'.charCodeAt(0), 2,];
    13	
    14	      case '"':
    15	      case '[':
    16	      case ']':
    17	        return [src.charCodeAt(pos + 1), 2,];
    18	      case '\\':
    19	        const codePoint = src.codePointAt(pos + 1);
    20	        if (codePoint === undefined) {
    21	          throw new GrammarParseError(src, pos, "Could not get code point for character");
    22	        }
    23	        return [codePoint, 2,];
    24	      default:
    25	        throw new GrammarParseError(src, pos, `Unknown escape at ${src[pos]}`);
    26	    }
    27	  }
    28	
    29	  if (!src[pos]) {
    30	    throw new GrammarParseError(src, pos, "Unexpected end of grammar input, failed to complete parse");
    31	  }
    32	  return [src.charCodeAt(pos), 1,];

     1	import { GrammarParseError, } from "../utils/errors/grammar-parse-error.js";
     2	import { isWordChar, } from "./is-word-char.js";
     3	import { parseChar, } from "./parse-char.js";
     4	import { parseName, } from "./parse-name.js";
     5	import { parseSpace, } from "./parse-space.js";
     6	import { SymbolIds, } from "./symbol-ids.js";
     7	import { InternalRuleType, InternalRuleDef, } from "./types.js";
     8	
     9	export class RulesBuilder {
    10	  private pos = 0;
    11	  symbolIds: SymbolIds = new SymbolIds();
    12	  rules: InternalRuleDef[][];
    13	  src: string;
    14	  start: number = performance.now();
    15	  #timeLimit: number;
    16	  constructor(src: string, limit = 1000) {
    17	    this.#timeLimit = limit;
    18	    this.rules = [];
    19	    this.src = src;
    20	    this.parse(src);
    21	  }
    22	
    23	  private parse = (src: string) => {
    24	    this.pos = parseSpace(src, 0, true); // move cursor forward until we reach non-whitespace content
    25	
    26	    while (this.pos < src.length) {
    27	      this.parseRule(src);
    28	    }
    29	
    30	
    31	    // Validate the state to ensure that all rules are defined
    32	    for (const rule of this.rules) {
    33	      for (const elem of rule) {
    34	        if (elem.type === InternalRuleType.RULE_REF) {
    35	          // Ensure that the rule at that location exists
    36	          const ruleExists = elem.value < this.rules.length && this.rules[elem.value]?.length > 0;
    37	          if (!ruleExists) {
    38	            const missingRuleName = this.symbolIds.reverseGet(elem.value);
    39	            let missingRulePos = this.symbolIds.getPos(missingRuleName);
    40	
    41	            // Skip over the ::= and any whitespace
    42	            while (missingRulePos < src.length && (src[missingRulePos] === ':' || src[missingRulePos] === '=' || /\s/.test(src[missingRulePos]))) {
    43	              missingRulePos++;
    44	            }
    45	
    46	            throw new GrammarParseError(src, missingRulePos, `Undefined rule identifier "${missingRuleName}"`);
    47	          }
    48	        }
    49	      }
    50	    }
    51	  };
    52	
    53	  private parseRule = (src: string): void => {
    54	    const name = parseName(src, this.pos);
    55	    this.pos = parseSpace(src, this.pos + name.length, false);
    56	    const ruleId: number = this.getSymbolId(name, name.length);
    57	
    58	    // Skip over whitespace characters and find the ::= sequence
    59	    this.pos = parseSpace(src, this.pos, true);
    60	    if (!(src[this.pos] === ':' && src[this.pos + 1] === ':' && src[this.pos + 2] === '=')) {
    61	      throw new GrammarParseError(src, this.pos, `Expecting ::= at ${this.pos}`);
    62	    }
    63	    this.pos = parseSpace(src, this.pos + 3, true);
    64	
    65	    this.parseAlternates(name, ruleId);
    66	
    67	    if (src[this.pos] === '\r') {
    68	      this.pos += src[this.pos + 1] === '\n' ? 2 : 1;
    69	    } else if (src[this.pos] === '\n') {
    70	      this.pos += 1;
    71	    } else if (src[this.pos]) {
    72	      throw new GrammarParseError(src, this.pos, `Expecting newline or end at ${this.pos}`);
    73	    }
    74	    this.pos = parseSpace(src, this.pos, true);
    75	  };
    76	
    77	  getSymbolId = (src: string, len: number): number => {
    78	    const nextId: number = this.symbolIds.size;
    79	    const key = src.slice(0, len);
    80	    if (!this.symbolIds.has(key)) {
    81	      this.symbolIds.set(key, nextId, this.pos);
    82	    }
    83	    return this.symbolIds.get(key);
    84	  };
    85	
    86	  generateSymbolId = (base_name: string): number => {
    87	    const next_id = this.symbolIds.size;
    88	    this.symbolIds.set(`${base_name}_${next_id}`, next_id, this.pos);
    89	    return next_id;
    90	  };
    91	
    92	  addRule = (
    93	    rule_id: number,
    94	    rule: InternalRuleDef[]
    95	  ): void => {
    96	    this.rules[rule_id] = rule;
    97	  };
    98	
    99	  checkDuration() {
   100	    if (performance.now() - this.start > this.#timeLimit) {
   101	      throw new GrammarParseError(this.src, this.pos, `duration of ${this.#timeLimit} exceeded:`);
   102	    }
   103	  }
   104	
   105	  parseSequence = (
   106	    rule_name: string,
   107	    outElements: InternalRuleDef[],
   108	    depth = 0,
   109	  ): void => {
   110	    const isNested = depth !== 0;
   111	    const src = this.src;
   112	    let lastSymStart: number = outElements.length;
   113	    while (src[this.pos]) {
   114	      if (src[this.pos] === '"') {
   115	        this.pos += 1;
   116	        lastSymStart = outElements.length;
   117	        while (src[this.pos] !== '"') {
   118	          this.checkDuration();
   119	          const [value, incPos,] = parseChar(src, this.pos);
   120	          outElements.push({ type: InternalRuleType.CHAR, value: [value,], });
   121	          this.pos += incPos; // Adjusting pos by the length of parsed characters
   122	        }
   123	        this.pos = parseSpace(src, this.pos + 1, isNested);
   124	      } else if (src[this.pos] === '[') {
   125	        this.pos += 1;
   126	        let startType: InternalRuleType = InternalRuleType.CHAR;
   127	        if (src[this.pos] === '^') {
   128	          this.pos += 1;
   129	          startType = InternalRuleType.CHAR_NOT;
   130	        }
   131	        lastSymStart = outElements.length;
   132	        while (src[this.pos] !== ']') {
   133	          this.checkDuration();
   134	          const type = lastSymStart < outElements.length ? InternalRuleType.CHAR_ALT : startType;
   135	          const [startcharValue, incPos,] = parseChar(src, this.pos);
   136	          this.pos += incPos;
   137	          if (type === InternalRuleType.CHAR || type === InternalRuleType.CHAR_NOT) {
   138	            outElements.push({ type, value: [startcharValue,], });
   139	          } else {
   140	            outElements.push({ type, value: startcharValue, });
   141	          }
   142	
   143	          if (src[this.pos] === '-' && src[this.pos + 1] !== ']') {
   144	            this.pos += 1;
   145	            const [endcharValue, incPos,] = parseChar(src, this.pos);
   146	            outElements.push({ type: InternalRuleType.CHAR_RNG_UPPER, value: endcharValue, });
   147	            this.pos += incPos;
   148	          }
   149	        }
   150	        this.pos = parseSpace(src, this.pos + 1, isNested);
   151	      } else if (isWordChar(src[this.pos])) {
   152	        const name = parseName(src, this.pos);
   153	        const refRuleId: number = this.getSymbolId(name, name.length);
   154	        this.pos += name.length;
   155	        this.pos = parseSpace(src, this.pos, isNested);
   156	
   157	        lastSymStart = outElements.length;
   158	        outElements.push({ type: InternalRuleType.RULE_REF, value: refRuleId, });
   159	      } else if (src[this.pos] === '(') {
   160	        this.pos = parseSpace(src, this.pos + 1, true);
   161	        const subRuleId: number = this.generateSymbolId(rule_name);
   162	        this.parseAlternates(rule_name, subRuleId, depth + 1);
   163	        lastSymStart = outElements.length;
   164	        outElements.push({ type: InternalRuleType.RULE_REF, value: subRuleId, });
   165	        if (src[this.pos] !== ')') {
   166	          throw new GrammarParseError(src, this.pos, `Expecting ')' at ${this.pos}`);
   167	        }
   168	        this.pos = parseSpace(src, this.pos + 1, isNested);
   169	      } else if (src[this.pos] === '*' || src[this.pos] === '+' || src[this.pos] === '?') {
   170	        if (lastSymStart === outElements.length) {
   171	          throw new GrammarParseError(src, this.pos, `Expecting preceding item to */+/? at ${this.pos}`);
   172	        }
   173	        const subRuleId: number = this.generateSymbolId(rule_name);
   174	        const subRule: InternalRuleDef[] = outElements.slice(lastSymStart);
   175	        if (src[this.pos] === '*' || src[this.pos] === '+') {
   176	          subRule.push({ type: InternalRuleType.RULE_REF, value: subRuleId, });
   177	        }

     1	export const MAXIMUM_NUMBER_OF_ERROR_LINES_TO_SHOW = 3;
     2	
     3	export const buildErrorPosition = (src: string, pos: number): string[] => {
     4	  if (src === "") {
     5	    return ["No input provided",];
     6	  }
     7	
     8	  const lines = src.split('\n');
     9	
    10	  let lineIdx = 0;
    11	  while (lines[lineIdx] && pos > lines[lineIdx].length - 1) {
    12	    pos -= lines[lineIdx].length;
    13	    lineIdx += 1;
    14	  }
    15	
    16	  const linesToShow: string[] = [];
    17	  for (let i = Math.max(0, lineIdx - (MAXIMUM_NUMBER_OF_ERROR_LINES_TO_SHOW - 1)); i <= lineIdx; i++) {
    18	    linesToShow.push(lines[i]);
    19	  }
    20	  return [
    21	    ...linesToShow,
    22	    ' '.repeat(pos) + '^',
    23	  ];
    24	};
