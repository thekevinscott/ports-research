    11	        raise ValueError("grammar must be a string")
    12	
    13	    if not isinstance(initial_string, str):

    20	    if len(rules) == 0:
    21	        raise GrammarParseError(grammar, 0, "No rules were found")
    22	    if symbol_ids["root"] is None:
    23	        raise GrammarParseError(grammar, 0, "Grammar does not contain a 'root' symbol")
    24	    root_id: int = symbol_ids["root"]
    25	
    26	    stacked_rules = [build_rule_stack(rule) for rule in rules]
    27	    graph = Graph(grammar, stacked_rules, root_id)

    46	        stacked_rules: list[list[list[UnresolvedRule]]],
    47	        root_id: int,

    51	        self.previous_code_points = []
    52	        rule_refs: list[RuleRef] = []
    53	        unique_rules: dict[str, UnresolvedRule] = {}
    54	
    55	        for stack_id in range(len(stacked_rules)):
    56	            stack = stacked_rules[stack_id]
    57	            nodes: dict[int, GraphNode] = {}
    58	            for path_id in range(len(stack)):
    59	                path = stack[path_id]

    61	                for step_id in range(len(path) - 1, -1, -1):
    62	                    next_node: GraphNode | None = node
    63	                    rule = stack[path_id][step_id]
    64	                    unique_rules[get_serialized_rule_key(rule)] = rule
    65	                    if is_rule_ref(rule):

    67	                    # rules coming in may be identical but have different references.
    68	                    # here, we ensure we always use the same reference for an identical rule.
    69	                    # this makes future comparisons easier.

    71	                    if unique_rule is None:
    72	                        raise ValueError("Could not get unique rule")
    73	                    node = GraphNode(
    74	                        unique_rule,
    75	                        {
    76	                            "stackId": stack_id,
    77	                            "pathId": path_id,
    78	                            "stepId": step_id,
    79	                        },
    80	                        next_node,
    81	                    )
    82	
    83	                if node is None:
    84	                    raise ValueError("Could not get node")

    89	        if root_node is None:
    90	            raise ValueError(f"Root node not found for value: {root_id}")

    92	
    93	        for rule_ref in rule_refs:
    94	            referenced_nodes = set()
    95	            referenced_nodes.update(self.__get_root_node__(rule_ref.value).values())

   121	    def __parse__(self, current_pointers: Pointers, code_point: int) -> Pointers:
   122	        for rule, graph_pointers in self.__iterate_over_pointers__(current_pointers):
   123	            if is_rule_char(rule):
   124	                valid = False
   125	                for possible_code_point in rule.value:
   126	                    if valid is True:
   127	                        pass
   128	                    elif is_range(possible_code_point):
   129	                        if is_point_in_range(code_point, possible_code_point):
   130	                            valid = True
   131	                    elif code_point == possible_code_point:
   132	                        valid = True
   133	                self.__set_valid__(graph_pointers, valid)
   134	
   135	            elif is_rule_char_exclude(rule):
   136	                valid = True
   137	                for possible_code_point in rule.value:
   138	                    if valid is False:
   139	                        pass
   140	                    elif is_range(possible_code_point):
   141	                        if is_point_in_range(code_point, possible_code_point):
   142	                            valid = False
   143	                    else:
   144	                        if code_point == possible_code_point:
   145	                            valid = False
   146	                self.__set_valid__(graph_pointers, valid)
   147	
   148	            elif not is_rule_end(rule):
   149	                raise ValueError(f"Unsupported rule: {rule}")
   150	
   151	        # a pointer's id is the sum of its node's id and its parent's id chain.
   152	        # if two pointers share the same id, it means they point to the same node and have identical parent chains.
   153	        # for the purposes of walking the graph, we only need to keep one of them.
   154	        next_pointers = Pointers()
   155	        for current_pointer in current_pointers:
   156	            for unresolved_next_pointer in current_pointer.fetch_next():
   157	                for resolved_next_pointer in self.__resolve_pointer__(
   158	                    unresolved_next_pointer,
   159	                ):
   160	                    next_pointers.add(resolved_next_pointer)
   161	        return next_pointers

   163	    def __resolve_pointer__(
   164	        self,
   165	        unresolved_pointer: GraphPointer,
   166	    ) -> Iterable[ResolvedGraphPointer]:
   167	        for resolved_pointer in unresolved_pointer.resolve():
   168	            if is_rule_ref(resolved_pointer.node.rule):
   169	                raise ValueError(
   170	                    "Encountered a reference rule when building pointers to the graph",
   171	                )
   172	            if (
   173	                is_rule_end(resolved_pointer.node.rule)
   174	                and resolved_pointer.parent is not None
   175	            ):
   176	                raise ValueError(
   177	                    "Encountered an ending rule with a parent when building pointers to the graph",
   178	                )
   179	            yield resolved_pointer

   181	    def add(self, src: ValidInput, pointers: Pointers | None = None) -> Pointers:
   182	        if not isinstance(src, str):
   183	            raise ValueError("src must be a string in graph.add")
   184	        pointers = pointers or self.__get_initial_pointers__()
   185	
   186	        code_points = get_input_as_code_points(src)

   248	        for pointer in pointers:

   254	            if seen_rule is None:

    43	    paths: list[UnresolvedRule] = []

    46	
    47	    while idx < len(linear_rules):
    48	        rule_def = linear_rules[idx]
    49	        if is_rule_def_char(rule_def) or is_rule_def_char_not(rule_def):

    68	                        )

    72	                    char_rule.value.append(
    73	                        cast(
    74	                            Range,
    75	                            [prev_value, cast(InternalRuleDefCharRngUpper, rule).value],
    76	                        ),
    77	                    )

    79	                    rule = cast(InternalRuleDefCharAlt, rule)

   231	                    )

   236	                        InternalRuleDefReference(

   238	                        ),
   239	                    )

   246	                    InternalRuleDefReference(
   247	                        value=sub_rule_id,
   248	                    ),
   249	                ]
   250	                self.pos = parse_space(src, self.pos + 1, is_nested)
   251	            else:

   255	        src = self.src
